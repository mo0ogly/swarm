package main
import("encoding/json";"go/ast";"go/parser";"go/token";"os";"strconv";"strings";"regexp")
type Entry struct {File string `json:"file"`; Line int `json:"line"`; Function string `json:"function"`; Kind string `json:"kind"`; Value string `json:"value"`}
var worksTable=regexp.MustCompile("(?i)\\bfrom[[:space:]]+(?:works|\"works\"|`works`|\\[works\\])(?:[^[:alnum:]_]|$)")
func main(){var paths []string;if err:=json.NewDecoder(os.Stdin).Decode(&paths);err!=nil{panic(err)}; out:=[]Entry{};fs:=token.NewFileSet();for _,p:=range paths{f,err:=parser.ParseFile(fs,p,nil,0);if err!=nil{panic(err)};for _,d:=range f.Decls{fn,ok:=d.(*ast.FuncDecl);if !ok||fn.Body==nil{continue};ast.Inspect(fn.Body,func(n ast.Node)bool{call,ok:=n.(*ast.CallExpr);if !ok{return true};if sel,ok:=call.Fun.(*ast.SelectorExpr);ok&&(sel.Sel.Name=="get"||sel.Sel.Name=="list"){out=append(out,Entry{p,fs.Position(call.Pos()).Line,fn.Name.Name,"selector",sel.Sel.Name})};for _,arg:=range call.Args{lit,ok:=arg.(*ast.BasicLit);if !ok||lit.Kind!=token.STRING{continue};value,err:=strconv.Unquote(lit.Value);if err==nil&&strings.HasPrefix(strings.ToUpper(strings.TrimSpace(value)),"SELECT ")&&worksTable.MatchString(value){out=append(out,Entry{p,fs.Position(call.Pos()).Line,fn.Name.Name,"literal_sql",value})}};return true})}};if err:=json.NewEncoder(os.Stdout).Encode(out);err!=nil{panic(err)}}
