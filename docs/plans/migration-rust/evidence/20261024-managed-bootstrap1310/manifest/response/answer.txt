/// Validates a prior manifest against the canonicalized request and canonical paths.
/// Mirrors Go validatePreparedRepository minus the `rev-parse --verify oid^{commit}`
/// loop (actual git objects are checked separately by the caller).
fn validate_manifest(
    prior: &ManagedRepository,
    expected: &ManagedRepositoryRequest,
    storage: &str,
    source: &str,
    subdir: &str,
    request_digest: &str,
) -> Result<(), AgentLaunchError> {
    let included_matches = match (&prior.included, &expected.include) {
        (None, empty) if empty.is_empty() => true,
        (Some(prior_included), expected_include) => prior_included == expected_include,
        _ => false,
    };
    if prior.request_digest != request_digest
        || prior.storage != storage
        || prior.source != source
        || prior.project_subdir != subdir
        || !included_matches
    {
        return Err(AgentLaunchError::new(
            "manifeste du dépôt incompatible avec la sélection et ses chemins canoniques ; réconcilier avant reprise",
        ));
    }
    let snapshot = hex::decode(&prior.snapshot_digest)
        .map_err(|_| AgentLaunchError::new("empreinte SHA-256 du dépôt préparé absente ou invalide"))?;
    if snapshot.len() != 32 {
        return Err(AgentLaunchError::new(
            "empreinte SHA-256 du dépôt préparé absente ou invalide",
        ));
    }
    for oid in [&prior.base_commit, &prior.candidate_commit] {
        if oid.is_empty() {
            return Err(AgentLaunchError::new(
                "identité Git du dépôt préparé absente",
            ));
        }
    }
    Ok(())
}
