/// filepath.Clean on an absolute path, before resolving symlinks (Linux qualified).
fn clean_absolute(path: &Path) -> PathBuf {
    let mut out = PathBuf::new();
    for component in path.components() {
        match component {
            Component::CurDir => {}
            Component::ParentDir => {
                out.pop();
            }
            other => out.push(other.as_os_str()),
        }
    }
    out
}
/// Go filepath.EvalSymlinks' technical traversal bound, not an operator policy.
pub const GO_SYMLINK_TRAVERSAL_LIMIT: usize = 255;
#[cfg(unix)]
fn resolve_symlinks(path: &Path) -> io::Result<PathBuf> {
    use std::{
        collections::VecDeque,
        ffi::OsString,
        os::unix::ffi::{OsStrExt, OsStringExt},
    };
    let mut pending: VecDeque<Vec<u8>> = path
        .as_os_str()
        .as_bytes()
        .split(|b| *b == b'/')
        .map(<[u8]>::to_vec)
        .collect();
    let mut dest = PathBuf::from("/");
    let mut links = 0usize;
    while let Some(part) = pending.pop_front() {
        if part.is_empty() || part == b"." {
            continue;
        }
        if part == b".." {
            dest.pop();
            if dest.as_os_str().is_empty() {
                dest.push("/");
            }
            continue;
        }
        dest.push(OsString::from_vec(part));
        let metadata = std::fs::symlink_metadata(&dest)?;
        if !metadata.file_type().is_symlink() {
            if !metadata.is_dir() && !pending.is_empty() {
                return Err(io::Error::from(io::ErrorKind::NotADirectory));
            }
            continue;
        }
        links += 1;
        if links > GO_SYMLINK_TRAVERSAL_LIMIT {
            return Err(io::Error::other("Go EvalSymlinks traversal limit exceeded"));
        }
        let target = std::fs::read_link(&dest)?;
        dest.pop();
        let bytes = target.as_os_str().as_bytes();
        if bytes.starts_with(b"/") {
            dest = PathBuf::from("/");
        }
        for component in bytes.split(|b| *b == b'/').rev() {
            pending.push_front(component.to_vec());
        }
    }
    Ok(dest)
}
#[cfg(not(unix))]
fn resolve_symlinks(path: &Path) -> io::Result<PathBuf> {
    std::fs::canonicalize(path)
}
pub fn canonical_project_root(root: &Path) -> Result<PathBuf, LocalPathError> {
    let absolute = if root.is_absolute() {
        root.to_path_buf()
    } else {
        std::env::current_dir()
            .map_err(|e| io_error(PathOperation::WorkingDirectory, e))?
            .join(root)
    };
    resolve_symlinks(&clean_absolute(&absolute))
        .map_err(|e| io_error(PathOperation::CanonicalRoot, e))
}
pub fn local_file(root: &Path, name: &str) -> Result<PathBuf, LocalPathError> {
    if !crate::text::nonempty(name) || Path::new(name).is_absolute() {
        return Err(LocalPathError::Refused(PathRefusal::RelativeRequired));
    }
    let root = canonical_project_root(root)?;
    let resolved = resolve_symlinks(&clean_absolute(&root.join(name)))
        .map_err(|e| io_error(PathOperation::ResolveFile, e))?;
    if !resolved.starts_with(&root) {
        return Err(LocalPathError::Refused(PathRefusal::OutsideProject));
    }
    if !std::fs::metadata(&resolved)
        .map_err(|e| io_error(PathOperation::FileMetadata, e))?
        .is_file()
    {
        return Err(LocalPathError::Refused(PathRefusal::RegularFileRequired));
    }
    Ok(resolved)
}
/// Go lexical cleanup before physical resolution; observation, not confinement.
pub fn local_directory(root: &Path, name: &str) -> Result<PathBuf, LocalPathError> {
    if Path::new(name).is_absolute() {
        return Err(LocalPathError::Refused(PathRefusal::RelativeRequired));
    }
    let root = canonical_project_root(root)?;
    let resolved = resolve_symlinks(&clean_absolute(&root.join(name)))
        .map_err(|e| io_error(PathOperation::ResolveFile, e))?;
    if !resolved.starts_with(&root) {
        return Err(LocalPathError::Refused(PathRefusal::OutsideProject));
    }
    if !std::fs::metadata(&resolved)
        .map_err(|e| io_error(PathOperation::FileMetadata, e))?
        .is_dir()
    {
        return Err(LocalPathError::Refused(PathRefusal::DirectoryRequired));
    }
    Ok(resolved)
}
pub fn canonical_owned_project_path(root: &Path, path: &Path) -> Result<PathBuf, LocalPathError> {
    let canonical = canonical_project_root(root)?;
    if !path.is_absolute() || clean_absolute(path).as_os_str() != path.as_os_str() {
        return Err(LocalPathError::Refused(
            PathRefusal::AbsoluteNormalizedRequired,
        ));
    }
    let ancestors: Vec<_> = path.ancestors().collect();
    for prefix in ancestors.into_iter().rev() {
        if resolve_symlinks(prefix).is_ok_and(|p| p == canonical) {
            let relative = path
                .strip_prefix(prefix)
                .map_err(|_| LocalPathError::Refused(PathRefusal::OutsideProject))?;
            let actual =
                resolve_symlinks(path).map_err(|e| io_error(PathOperation::ResolveOwned, e))?;
            if actual != canonical.join(relative) {
                return Err(LocalPathError::Refused(PathRefusal::RedirectedOwnedPath));
            }
            return Ok(actual);
        }
    }
    Err(LocalPathError::Refused(PathRefusal::OutsideProject))
}

			return nil, err
		}
		var prior ManagedRepository
		if err = json.Unmarshal(raw, &prior); err != nil {
			return nil, fmt.Errorf("manifeste préparé invalide, conservé pour diagnostic : %w", err)
		}
		if err = validatePreparedRepository(prior, storage, source, subdir, digest, r.Include); err != nil {
			return nil, fmt.Errorf("preuve du dépôt préparé conservée sans réécriture : %w", err)
		}
		encoded, err := json.Marshal(prior)
		if err != nil {
			return nil, err
		}
		if retained != nil && string(canonical) != string(encoded) {
			return nil, fmt.Errorf("manifestes final et préparé divergents ; conserver les originaux et réconcilier avant reprise")
		}
		if retained == nil {
			retained, retainedRaw, canonical = &prior, raw, encoded
		}
		finalPresent = finalPresent || i == 0
	}
	if retained != nil {
		if !finalPresent {
			if err = atomicWrite(manifest, retainedRaw); err != nil {
				return nil, err
			}
		}
		return retained, nil
	}
	base, err := managedGit(source, "rev-parse", "HEAD")
	if err != nil {
		return nil, err
	}
	dirty, err := managedGit(source, "status", "--porcelain", "--untracked-files=normal")
	if err != nil {
		return nil, err
	}
	if dirty != "" && !r.CommittedOnly && len(r.Include) == 0 {
		return nil, fmt.Errorf("dépôt modifié : choisir le commit enregistré ou nommer les fichiers de l’instantané")
	}
	if r.CommittedOnly && len(r.Include) > 0 {
		return nil, fmt.Errorf("choisir le commit seul ou une liste de fichiers, pas les deux")
	}
	staging, err := os.MkdirTemp(storage, "repository-build-")
	if err != nil {
		return nil, err
	}
	defer os.RemoveAll(staging)
	bare := filepath.Join(staging, "repository.git")
	if _, err = managedGit(storage, "clone", "--bare", "--no-local", "--no-hardlinks", top, bare); err != nil {
		return nil, err
	}
	included := map[string]string{}
	if len(r.Include) > 0 {
		copyDir := filepath.Join(staging, "snapshot")
		if _, err = managedGit(storage, "clone", "--no-local", bare, copyDir); err != nil {
			return nil, err
		}
		if _, err = managedGit(copyDir, "checkout", "--detach", base); err != nil {
			return nil, err
		}
		total := int64(0)
		for _, name := range r.Include {
			if name == "" || filepath.IsAbs(name) || filepath.Clean(name) != name || name == ".." || strings.HasPrefix(name, "../") || strings.HasPrefix(name, ".git/") || strings.HasPrefix(name, ".swarm/") || name == ".git" || name == ".swarm" {
				return nil, fmt.Errorf("chemin d’instantané interdit : %s", name)
			}
			if _, err = managedGit(source, "check-ignore", "--", name); err == nil {
				return nil, fmt.Errorf("fichier ignoré exclu de l’instantané : %s", name)
			}
			src := filepath.Join(source, name)
			dest := filepath.Join(copyDir, subdir, name)
			for parent := dest; parent != copyDir; parent = filepath.Dir(parent) {
				if st, e := os.Lstat(parent); e == nil && st.Mode()&os.ModeSymlink != 0 {
					return nil, fmt.Errorf("lien interdit dans la destination : %s", name)
				}
			}
			stat, statErr := os.Lstat(src)
			if os.IsNotExist(statErr) {
				if _, err = managedGit(source, "ls-files", "--error-unmatch", "--", name); err != nil {
					return nil, err
				}
				if err = os.Remove(dest); err != nil && !os.IsNotExist(err) {
					return nil, err
				}
				included[name] = "deleted"
				continue
			}
			if statErr != nil || !stat.Mode().IsRegular() {
				return nil, fmt.Errorf("fichier régulier requis : %s", name)
			}
			if _, err = localFile(source, name); err != nil {
				return nil, err
			}
			total += stat.Size()
			if total > 64<<20 {
				return nil, fmt.Errorf("instantané supérieur à 64 Mio")
			}
			data, err := os.ReadFile(src)
			if err != nil {
				return nil, err
			}
			if err = os.MkdirAll(filepath.Dir(dest), 0700); err != nil {
				return nil, err
			}
			if err = os.WriteFile(dest, data, stat.Mode().Perm()); err != nil {
				return nil, err
			}
			included[name] = hash(data)
		}
		gitPaths := []string{}
		for _, name := range r.Include {
			gitPaths = append(gitPaths, filepath.Join(subdir, name))
		}
		if _, err = managedGit(copyDir, append([]string{"add", "-A", "--"}, gitPaths...)...); err != nil {
			return nil, err
		}
		tree, err := managedGit(copyDir, "write-tree")
		if err != nil {
			return nil, err
		}
		base, err = managedGit(copyDir, "commit-tree", tree, "-p", base, "-m", "Instantané autorisé pour Swarm")
		if err != nil {
			return nil, err
		}
		if _, err = managedGit(bare, "fetch", "--no-tags", copyDir, base); err != nil {
			return nil, err
		}
	}
	if _, err = managedGit(bare, "update-ref", "refs/heads/swarm-result", base); err != nil {
		return nil, err
	}
	if _, err = managedGit(bare, "symbolic-ref", "HEAD", "refs/heads/swarm-result"); err != nil {
		return nil, err
	}
	final := filepath.Join(storage, "repository.git")
	if _, err = os.Lstat(final); err == nil {
		return nil, fmt.Errorf("dépôt préparé sans manifeste : examiner avant reprise")
	}
	snap, _ := json.Marshal(map[string]any{"base": base, "files": included})
	repo := &ManagedRepository{Subdir: subdir, Source: source, Storage: storage, Base: base, Candidate: base, Snapshot: hash(snap), Included: r.Include, RequestDigest: digest}
	raw, _ := json.Marshal(repo)
	if err = atomicWrite(manifest+".pending", raw); err != nil {
		return nil, err
	}
	if err = os.Rename(bare, final); err != nil {
		return nil, err
	}
	if err = atomicWrite(manifest, raw); err != nil {
		return nil, err
	}
	return repo, nil
}
func validatePreparedRepository(prior ManagedRepository, storage, source, subdir, digest string, include []string) error {
	if prior.RequestDigest != digest || prior.Storage != storage || prior.Source != source || prior.Subdir != subdir || !slices.Equal(prior.Included, include) {
		return fmt.Errorf("manifeste du dépôt incompatible avec la sélection et ses chemins canoniques ; réconcilier avant reprise")
	}
	snapshot, err := hex.DecodeString(prior.Snapshot)
	if err != nil || len(snapshot) != 32 {
		return fmt.Errorf("empreinte SHA-256 du dépôt préparé absente ou invalide")
	}
	for _, oid := range []string{prior.Base, prior.Candidate} {
		if oid == "" {
			return fmt.Errorf("identité Git du dépôt préparé absente")
		}
		resolved, err := managedGit(filepath.Join(storage, "repository.git"), "rev-parse", "--verify", oid+"^{commit}")
		if err != nil {
			return fmt.Errorf("objet Git préparé absent : %w", err)
		}
		if resolved != oid {
			return fmt.Errorf("identité Git préparée non canonique")
		}
	}
	return nil
}
