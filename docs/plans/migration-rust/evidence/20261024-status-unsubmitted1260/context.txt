//! Real active/terminal report discovery and Go fallback diagnostics. No
//! provider/process/SQL writes or acceptance. Reports are filenames for inspection,
//! never evidence of completion. No Go5000-entry silent search cutoff is copied.
use super::{finished, RootStoreError as E};
use crate::{Agent, AttemptDiagnostic, DiagnosticItem};
use rustix::fs::{self, AtFlags, FileType, Mode, OFlags};
use std::{
    error::Error,
    fs::File,
    os::{fd::IntoRawFd, unix::fs::MetadataExt},
    path::Path,
};

/// Preserve the original typed IO/encoding cause behind the safe existing error.
fn cause(code: &'static str, cause: impl Error + Send + Sync + 'static) -> E {
    E::Mutation(crate::WorkEngineError::Mission(Box::new(
        crate::MissionSupervisionError::Cause {
            code,
            cause: Box::new(cause),
        },
    )))
}
/// Close the owned directory explicitly, preserving primary and close causes.
fn close<T>(result: Result<T, E>, file: File) -> Result<T, E> {
    finished(
        result,
        nix::unistd::close(file.into_raw_fd())
            .map_err(|e| cause("mission-status-reports-close", e)),
    )
}
/// Require an actual opened directory, allowing every OS-readable owner/mode.
/// Project-private datastore ownership remains the outer Root guard's contract.
fn directory(file: &File) -> Result<std::fs::Metadata, E> {
    let metadata = file
        .metadata()
        .map_err(|e| cause("mission-status-reports-metadata", e))?;
    if !metadata.is_dir() {
        return Err(E::Refused("mission-status-reports-not-directory"));
    }
    Ok(metadata)
}
/// Walk descriptor-anchored directories without following directory symlinks.
/// Selected file symlinks use the shared contained regular-artifact reader, so
/// an OS-readable regular target inside Root is accepted regardless of owner.
/// All selected filenames/entry-mtimes are retained as Go DirEntry.Info does
/// (lstat for a symlink). Target metadata establishes regular containment only;
/// no contents are read or trusted and content-read permission is unnecessary.
/// Directory device/inode/mtime/ctime checks detect observed traversal changes;
/// they are sequential observations, not an atomic snapshot or ABA protection.
fn walk(
    root: &Path,
    file: &File,
    prefix: &str,
    task: &str,
    found: &mut Vec<(String, i128)>,
) -> Result<(), E> {
    let before = directory(file)?;
    let mut buffer = vec![
        std::mem::MaybeUninit::<u8>::uninit();
        usize::from(u16::MAX) + 1 + std::mem::align_of::<u64>()
    ];
    let mut reader = fs::RawDir::new(file, &mut buffer);
    while let Some(entry) = reader.next() {
        let entry = entry.map_err(|e| cause("mission-status-reports-read", e))?;
        let raw = entry.file_name().to_bytes();
        if matches!(raw, b"." | b"..") {
            continue;
        }
        let name = std::str::from_utf8(raw).map_err(|e| cause("mission-status-reports-name", e))?;
        let selected = name == format!("{task}.md")
            || (name.starts_with(&format!("{task}-"))
                && name.contains("handoff")
                && name.ends_with(".md"));
        let stat = fs::statat(file, entry.file_name(), AtFlags::SYMLINK_NOFOLLOW)
            .map_err(|e| cause("mission-status-reports-stat", e))?;
        let kind = FileType::from_raw_mode(stat.st_mode);
        let path = format!("{prefix}/{name}");
        if kind == FileType::Directory {
            let child: File = fs::openat(
                file,
                entry.file_name(),
                OFlags::RDONLY | OFlags::DIRECTORY | OFlags::NOFOLLOW | OFlags::CLOEXEC,
                Mode::empty(),
            )
            .map_err(|e| cause("mission-status-reports-open-directory", e))?
            .into();
            let result = (|| {
                let actual = directory(&child)?;
                if actual.dev() != stat.st_dev || actual.ino() != stat.st_ino {
                    return Err(E::Refused("mission-status-reports-directory-changed"));
                }
                walk(root, &child, &path, task, found)?;
                let after = fs::statat(file, entry.file_name(), AtFlags::SYMLINK_NOFOLLOW)
                    .map_err(|e| cause("mission-status-reports-restat", e))?;
                if stat.st_dev != after.st_dev || stat.st_ino != after.st_ino {
                    return Err(E::Refused("mission-status-reports-directory-changed"));
                }
                Ok(())
            })();
            close(result, child)?;
        } else if selected {
            let (_, metadata) = crate::gate_evaluation::artifact_metadata(root, &path)
                .map_err(|cause| E::Gate(crate::GateContractError::Evaluation(cause)))?;
            let after = fs::statat(file, entry.file_name(), AtFlags::SYMLINK_NOFOLLOW)
                .map_err(|e| cause("mission-status-reports-restat", e))?;
            if (
                stat.st_dev,
                stat.st_ino,
                stat.st_mode,
                stat.st_mtime,
                stat.st_mtime_nsec,
                stat.st_ctime,
                stat.st_ctime_nsec,
            ) != (
                after.st_dev,
                after.st_ino,
                after.st_mode,
                after.st_mtime,
                after.st_mtime_nsec,
                after.st_ctime,
                after.st_ctime_nsec,
            ) {
                return Err(E::Refused("mission-status-reports-file-changed"));
            }
            // A regular source path and its checked target must name the same
            // inode. For a symlink, local_file supplies the containment check.
            if kind == FileType::RegularFile
                && (metadata.dev() != stat.st_dev || metadata.ino() != stat.st_ino)
            {
                return Err(E::Refused("mission-status-reports-file-changed"));
            }
            found.push((
                path,
                i128::from(stat.st_mtime) * 1_000_000_000 + i128::from(stat.st_mtime_nsec),
            ));
        }
    }
    let after = directory(file)?;
    let signature = |m: &std::fs::Metadata| {
        (
            m.dev(),
            m.ino(),
            m.mtime(),
            m.mtime_nsec(),
            m.ctime(),
            m.ctime_nsec(),
        )
    };
    if signature(&before) != signature(&after) {
        return Err(E::Refused("mission-status-reports-directory-changed"));
    }
    Ok(())
}
/// Observe actual docs reports, newest mtime first with lexical tie order. ENOENT
/// for docs alone proves an empty listing. Contained regular file symlinks are
/// included; directory symlinks are never traversed. Directory/file owners need
/// not match the process. Permission, outside-Root/nonregular targets, child
/// races, IO and explicit close failures preserve their causes and refuse.
/// No directory is created and no report content or attribution is inferred.
pub(super) fn task_reports(root: &Path, task: &str) -> Result<Vec<String>, E> {
    if !swarm_identity_contract::name(task.as_bytes()) {
        return Err(E::Refused("mission-status-report-task-invalid"));
    }
    let path = root.join("docs");
    let file: File = match fs::open(
        &path,
        OFlags::RDONLY | OFlags::DIRECTORY | OFlags::NOFOLLOW | OFlags::CLOEXEC,
        Mode::empty(),
    ) {
        Ok(fd) => fd.into(),
        Err(rustix::io::Errno::NOENT) => return Ok(Vec::new()),
        Err(e) => return Err(cause("mission-status-reports-open", e)),
    };
    let result = (|| {
        let before = directory(&file)?;
        let mut found = Vec::new();
        walk(root, &file, "docs", task, &mut found)?;
        let current = std::fs::symlink_metadata(&path)
            .map_err(|e| cause("mission-status-reports-path", e))?;
        if !current.is_dir() || before.dev() != current.dev() || before.ino() != current.ino() {
            return Err(E::Refused("mission-status-reports-directory-changed"));
        }
        found.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
        Ok(found.into_iter().map(|(path, _)| path).collect())
    })();
    close(result, file)
}
/// Go operationText masking and terminal-cell clipping for fallback traces.
/// Sanitization and sensitivity masking precede the Go-compatible 600-cell
/// representation bound. Whole clusters are preserved; checked width overflow
/// retains its typed cause. Pure presentation, no execution/permission policy.
fn trace(reason: &str) -> Result<String, E> {
    let text = crate::agent_terminal_text::terminal_text(reason);
    let lower = text.to_lowercase();
    if [
        "password",
        "passwd",
        "secret",
        "token",
        "api_key",
        "api-key",
        "authorization",
        "private key",
    ]
    .iter()
    .any(|m| lower.contains(m))
    {
        return Ok("[détail masqué : donnée potentiellement sensible]".into());
    }
    crate::terminal_clip::clip(&text, 600)
        .map_err(|error| cause("mission-status-diagnostic-width", error))
}
/// Go failureCategory over the actual retained terminal cause, never task prose.
fn category(reason: &str) -> &'static str {
    let text = reason.to_lowercase();
    if [
        "permission denied",
        "operation not permitted",
        "read-only file system",
        "sandbox denied",
        "sandbox violation",
        "mount failed",
        "bwrap: can't bind mount",
        "unable to apply mount flags",
        "mount: permission denied",
        "cifs error",
        "socket: operation not permitted",
        "network is unreachable",
    ]
    .iter()
    .any(|m| text.contains(m))
    {
        "environment"
    } else if [
        "go test",
        "pytest",
        "unittest",
        "npm test",
        "node --test",
        "diff --check",
        "lint",
        "check_",
        "check-",
        "validation",
    ]
    .iter()
    .any(|m| text.contains(m))
    {
        "check"
    } else if [
        "not configured",
        "non configur",
        "configuration",
        "executable",
        "exécutable",
        "command not found",
        "no such file or directory",
    ]
    .iter()
    .any(|m| text.contains(m))
    {
        "configuration"
    } else {
        "unknown"
    }
}
/// Construct the single real fallback item (no structured tool failure supplied).
/// The text/category is the Go template, not a new execution policy or diagnosis.
fn item(category: &str, reason: &str, trace: String) -> DiagnosticItem {
    let (label,cause,consequence,action,kind)=match category {
        "configuration"=>("Configuration","La configuration nécessaire au lancement ou à l’outil est absente, invalide ou indisponible.","L’étape concernée n’a pas pu démarrer ou s’exécuter ; son résultat n’est pas démontré.","Corriger la configuration indiquée, vérifier qu’elle est relue, puis demander explicitement une nouvelle tentative.","correct_configuration"),
        "environment"=>("Environnement d’exécution","Les traces signalent un refus d’accès ou une ressource indisponible dans l’environnement.","La tentative ne peut pas prouver le résultat dans cet environnement ; répéter sans changement de précondition risque le même échec.","Faire vérifier les droits et l’accès aux ressources sur la machine qui exécute l’agent ; reprendre après correction vérifiée.","verify_environment"),
        "check"=>("Contrôle en échec","Un test ou contrôle observable a échoué.","Le critère couvert par ce contrôle reste non validé.","Examiner la trace, corriger la cause, puis rejouer le même contrôle avec les mêmes options avant toute validation.","fix_and_rerun_check"),
        "limit"=>("Limite atteinte",reason,"Le superviseur a interrompu la tentative ; cette interruption ne valide ni le livrable ni la tâche.","Examiner les erreurs précédentes et les préconditions ; reprendre explicitement sans relever arbitrairement les protections.","review_limit"),
        _=>("Cause inconnue","La cause exacte n’est pas disponible dans les événements structurés de cette tentative.","Le résultat de la tentative reste indéterminé ; aucune reprise automatique sûre ne peut être déduite.","Ouvrir les traces conservées et établir la cause avant de choisir une reprise.","inspect_traces"),
    };
    DiagnosticItem {
        category: category.into(),
        label: label.into(),
        count: 1,
        cause: cause.into(),
        consequence: consequence.into(),
        action: action.into(),
        action_kind: kind.into(),
        unknown: category == "unknown",
        traces: Some(vec![trace]),
    }
}
/// Produce Go fallbackAttemptDiagnostic from actual stored terminal fields.
/// Existing structured diagnostics take precedence. Explicit operator interruption
/// produces Go's empty diagnostic value; completed Agents without structured
/// diagnostics also produce Go's empty value, regardless of activity prose.
/// Failed/nonoperator-interrupted causes receive the real fallback.
/// Fallback traces use whole-cluster cell clipping; width overflow retains its cause.
pub(super) fn terminal_diagnostics(agents: &[Agent]) -> Result<Vec<AttemptDiagnostic>, E> {
    agents
        .iter()
        .map(|agent| {
            if !matches!(
                agent.status.as_str(),
                "failed" | "interrupted" | "completed"
            ) {
                return Err(E::Refused("mission-status-agent-classification-unported"));
            }
            if agent
                .diagnostic
                .items
                .as_ref()
                .is_some_and(|v| !v.is_empty())
                || agent.diagnostic.observed_errors > 0
            {
                return Ok(agent.diagnostic.clone());
            }
            if agent.status == "completed"
                || (agent.status == "interrupted" && agent.stop_kind == "operateur")
            {
                return Ok(AttemptDiagnostic::default());
            }
            let mut reason = crate::text::trim_space(&agent.activity);
            if reason.is_empty() {
                reason = "Tentative terminée sans motif technique enregistré";
            }
            let lower = reason.to_lowercase();
            let limited = ["limite", "délai", "budget de temps"]
                .iter()
                .any(|m| lower.contains(m));
            let consecutive = limited && lower.contains("erreurs d'outils consécutives");
            let category = if limited { "limit" } else { category(reason) };
            let item = item(category, &format!("{reason}."), trace(reason)?);
            let cap = agent.limits.max_consecutive_errors;
            let mut summary = "Aucune erreur d’outil structurée n’a été observée.".to_string();
            if cap > 0 {
                summary.push_str(&format!(
                    " Le plafond configuré est de {cap} erreurs consécutives"
                ));
                summary.push_str(if consecutive {
                    " et il a interrompu la tentative."
                } else {
                    " ; ce plafond est distinct du nombre total constaté."
                });
            }
            if limited && !consecutive {
                summary.push_str(&format!(
                    " Une autre limite d’exécution a interrompu la tentative : {reason}."
                ));
            }
            Ok(AttemptDiagnostic {
                agent_id: agent.id.clone(),
                attempt_id: agent.attempt_id.clone(),
                observed_errors: 0,
                consecutive_error_limit: cap,
                limit_reached: limited,
                consecutive_error_limit_reached: consecutive,
                limit_reason: if limited {
                    reason.into()
                } else {
                    String::new()
                },
                summary,
                items: Some(vec![item]),
            })
        })
        .collect()
}
/// Go recoveryCategoryFor environment branch and its terminal-limit precedence.
/// Other recovery categories are irrelevant to the supported stopped result path.
pub(super) fn environment_recovery(agent: &Agent, diagnostic: &AttemptDiagnostic) -> bool {
    agent.provider_cooldown.is_none()
        && !(agent.stop_kind == "garde" && diagnostic.limit_reached)
        && diagnostic.items.as_ref().is_some_and(|items| {
            items
                .iter()
                .any(|item| matches!(item.category.as_str(), "environment" | "configuration"))
        })
}

/// Preserve complete row order while observing active diagnostics without a
/// terminal fallback. Real structured items/errors are retained; active activity
/// prose never fabricates a failure. Nonactive rows use the strict terminal
/// producer, whose unsupported-state errors and original causes propagate.
pub(super) fn diagnostics(agents: &[Agent]) -> Result<Vec<AttemptDiagnostic>, E> {
    agents
        .iter()
        .map(|agent| {
            if agent.is_active() {
                Ok(
                    if agent
                        .diagnostic
                        .items
                        .as_ref()
                        .is_some_and(|v| !v.is_empty())
                        || agent.diagnostic.observed_errors > 0
                    {
                        agent.diagnostic.clone()
                    } else {
                        AttemptDiagnostic::default()
                    },
                )
            } else {
                terminal_diagnostics(std::slice::from_ref(agent)).map(|mut values| values.remove(0))
            }
        })
        .collect()
}
/// Supported nonmanaged planned lifecycle pairs, never liveness or permission.
/// Full recorded history must agree with its task; Root separately validates
/// original rows, organization, supported review shape and current process observations.
/// No IO, mutation, hidden policy or fallback to success for unknown states.
fn planned_pair(task: &crate::Task, agent: &Agent) -> bool {
    task.id == agent.task_id && match task.status.as_str() {
        "running" => agent.is_active(),
        "blocked" => matches!(agent.status.as_str(),"failed" | "interrupted") || (agent.status=="completed" && matching_review(task,agent)),
        "submitted" | "accepted" => agent.status == "completed",
        _ => false,
    }
}


/// A non-passed current opinion requires the real review display producer.
/// Go skips passed opinions before projecting them; skipping proves
/// neither validity nor acceptance. Actual accepted freshness independently
/// rechecks current native originals, gate, report and dependency evidence.
fn review_requires_projection(task: &crate::Task) -> bool {
    task.independent_review.as_deref().is_some_and(|review| review.state != "passed")
}

/// Match the actual current attempt/producer for a non-passed opinion. A
/// matched row is not yet current evidence; the observed producer checks it.
fn matching_review(task:&crate::Task,agent:&Agent)->bool {
    task.independent_review.as_deref().is_some_and(|review| super::mission_status_review::applies(true,task.attempts.as_ref().and_then(|v|v.last()).map(|a|a.id.as_str()),&agent.id,&agent.attempt_id,review))
}

/// Bound the newly wired review branch to nonmanaged planning and a completed
/// current producer on submitted/blocked task. Earlier reviews/recovery are
/// still refused separately; this shape check never validates its artifacts.
fn review_shape(work:&crate::Work,task:&crate::Task,agents:&[Agent])->bool {
    !review_requires_projection(task) || (work.planning.is_some() && matches!(task.status.as_str(),"submitted"|"blocked") && agents.iter().find(|a|a.task_id==task.id).is_some_and(|a|a.status=="completed" && matching_review(task,a)))
}

/// Require only branches whose result/readiness producers are implemented here.
/// Retained profiles are accepted for stopped or completed/submitted results
/// because Go does not consume dispatch readiness in those branches. All Agent
/// rows must be terminal or active on an ordinary running task; earlier terminal
/// rows remain history. Active classification must be observed separately; a
/// saved running status alone proves no live process.
/// Submitted results without a recent Agent use retained task/gate facts and
/// unknown process/no Agent identity. Last retained Task attempt remains visible;
/// no readiness is consumed or invented.
/// Abandoned decisions without a matching Agent also bypass readiness; they
/// remain nonvalidated, retaining the same review/recovery refusals. Accepted
/// tasks with completed or no selected Agent consume actual gate/freshness facts;
/// they retain unsupported/prior review and recovery refusals. No acceptance follows from status
/// or from a terminal process alone. Abandoned
/// tasks with an Agent and completed results outside submitted/accepted still
/// refuse. Unstarted profiles require the separate real readiness observation;
/// review/recovery history on those profiled tasks remains refused. This read
/// never changes the recorded decision or grants a launch. Current non-passed
/// review shapes use matching latest attempt/producer and the real review reader.
/// Ready/waiting planning trees support claimed/renewed leases, empty decisions and
/// initial todo/abandoned tasks without attempt/review/recovery history after structural
/// organization validation. Missing review remains non-current, never acceptance. Holder/date pairing and parsed
/// lease shape are checked, never treated as liveness or launch authority. Saved
/// counters at their cap remain readable. Submitted nonmanaged tasks may retain
/// completed Agents and their actual attempts, with no prior review/recovery history.
/// Final passed opinions may fall through; accepted tasks consume the actual
/// freshness/gate reader and never become validated from an opinion alone.
/// Running tasks may retain queued/starting/running/stopping Agents; blocked
/// tasks may retain failed/interrupted Agents. Actual process observation stays
/// mandatory before a live classification, never inferred from these rows.
/// Completed but unsubmitted without a current review, recovery/managed execution
/// and closed scopes still refuse at this port boundary, not a quota. Full Agent history is required;
/// callers must validate before applying display windows. Root, parent,
/// ownership and depth invariants come from structural validation, not scope count.
/// Original validation causes survive; configuration establishes no executed review.
/// No Agent state, review, recovery or permission is inferred from missing data.
pub(super) fn supported(work: &crate::Work, agents: &[Agent]) -> Result<(), E> {
    if agents.iter().any(|a| {
        a.is_active()
            && !work
                .tasks
                .as_deref()
                .unwrap_or(&[])
                .iter()
                .any(|t| t.id == a.task_id && t.status == "running")
    }) {
        return Err(E::Refused("mission-status-active-task-mismatch"));
    }
    if let Some(planning) = work.planning.as_deref() {
        // Only the initial ready/waiting tree lifecycle is assembled. These shape
        // checks delimit the port, never a new execution policy or saved cap.
        let scopes = planning.scopes.as_deref().unwrap_or_default();
        if (planning.repository.is_some() && !agents.is_empty())
            || agents.iter().any(|agent|
                !work.tasks.iter().flatten().any(|task| planned_pair(task,agent)))
            || scopes.iter().any(|scope| !matches!(scope.state.as_str(), "ready" | "waiting"))
        {
            return Err(E::Refused("mission-status-planning-unported"));
        }
        if work.tasks.iter().flatten().any(|task|
            !matches!(task.status.as_str(),"todo" | "abandoned" | "submitted" | "accepted" | "running" | "blocked")
            || (matches!(task.status.as_str(),"submitted" | "accepted" | "running" | "blocked")
                && !agents.iter().any(|agent| planned_pair(task,agent)))
            || (!matches!(task.status.as_str(),"submitted" | "accepted" | "running" | "blocked")
                && task.attempts.as_ref().is_some_and(|v| !v.is_empty()))
            || !review_shape(work,task,agents)
            || task.previous_reviews.as_ref().is_some_and(|v| !v.is_empty())
            || task.corrective_recovery.is_some() || task.planning_retry
            || task.review_coordination.is_some() || task.batch_review_resume.is_some()
            || task.recovered_result.is_some())
        {
            return Err(E::Refused("mission-status-planning-initial-tasks-unported"));
        }
        for scope in scopes {
            if scope.holder.is_empty() != scope.lease_until.is_empty()
                || (!scope.holder.is_empty() && !swarm_identity_contract::name(scope.holder.as_bytes()))
            {
                return Err(E::Refused("mission-status-planning-lease-shape-invalid"));
            }
            if !scope.lease_until.is_empty() {
                crate::mission_supervision::parse(&scope.lease_until)
                    .map_err(|error| cause("mission-status-planning-lease-invalid", error))?;
            }
        }
        crate::validate_mission_organization(work).map_err(|cause| {
            E::Mutation(crate::WorkEngineError::Mission(Box::new(
                crate::MissionSupervisionError::Cause {
                    code: "mission-status-organization-invalid", cause: Box::new(cause),
                },
            )))
        })?;
    }
    if agents.iter().any(|a| {
        !matches!(
            a.status.as_str(),
            "failed" | "interrupted" | "completed" | "queued" | "starting" | "running" | "stopping"
        )
    }) {
        return Err(E::Refused("mission-status-agent-classification-unported"));
    }
    // Older terminal rows remain retained history. The authoritative reader is
    // latest-first; consuming task/result branches select their first matching
    // Agent, while every row still receives ownership/shape validation upstream.
    for task in work.tasks.as_deref().unwrap_or(&[]) {
        let agent = agents.iter().find(|a| a.task_id == task.id);
        if agent.is_some() || matches!(task.status.as_str(), "submitted" | "abandoned" | "accepted") {
            let supported_state = match agent.map(|a| a.status.as_str()) {
                Some("queued" | "starting" | "running" | "stopping") => task.status == "running",
                Some("completed") => matches!(task.status.as_str(), "submitted" | "accepted") || (task.status=="blocked" && agent.is_some_and(|a|matching_review(task,a))),
                Some("failed" | "interrupted") => task.status == "blocked",
                None => matches!(task.status.as_str(), "submitted" | "abandoned" | "accepted"),
                _ => false,
            };
            if !supported_state
                || !review_shape(work,task,agents)
                || task
                    .previous_reviews
                    .as_ref()
                    .is_some_and(|v| !v.is_empty())
                || task.corrective_recovery.is_some()
                || task.planning_retry
                || task.review_coordination.is_some()
                || task.batch_review_resume.is_some()
                || task.recovered_result.is_some()
            {
                return Err(E::Refused("mission-status-terminal-task-branch-unported"));
            }
        } else {
            if task.status != "todo" || task.attempts.as_ref().is_some_and(|v| !v.is_empty()) {
                return Err(E::Refused("mission-status-task-history-unported"));
            }
            if (task.launch_profile.is_some() || work.launch_profile.is_some())
                && (!review_shape(work,task,agents)
                    || task.previous_reviews.as_ref().is_some_and(|v| !v.is_empty())
                    || task.corrective_recovery.is_some() || task.planning_retry
                    || task.review_coordination.is_some() || task.batch_review_resume.is_some()
                    || task.recovered_result.is_some())
            {
                return Err(E::Refused("mission-status-profile-task-branch-unported"));
            }
        }
    }
    Ok(())
}
/// Check only projection prerequisites after full-history supported() succeeds.
/// A display window may hide completed submitted/accepted history, keeping its
/// identity unknown. Running/blocked states require a displayed matching Agent;
/// otherwise the existing task-history refusal prevents invented readiness.
/// No IO, mutation, policy/default or process/launch authority is established.
/// Full history is still required separately; this cannot replace eligibility.
pub(super) fn supported_view(work: &crate::Work, agents: &[Agent]) -> Result<(), E> {
    if work.tasks.iter().flatten().any(|task|
        !matches!(task.status.as_str(),"todo" | "submitted" | "accepted" | "abandoned")
        && !agents.iter().any(|agent| agent.task_id == task.id))
    {
        return Err(E::Refused("mission-status-task-history-unported"));
    }
    Ok(())
}
/// Borrow recorded Agent states and real diagnostics in identical row order.
/// Active rows require the caller to replace observed_state using fresh activity
/// observations before projection; stored status alone proves no liveness.
/// Every diagnostic must correspond positionally to its original complete row.
/// Terminal-limit/provider cooldown precedence follows recoveryCategoryFor.
pub(super) fn agent_observations<'a>(
    agents: &'a [Agent],
    diagnostics: &'a [AttemptDiagnostic],
) -> Result<Vec<crate::mission_status_tasks::AgentObservation<'a>>, E> {
    if agents.len() != diagnostics.len() {
        return Err(E::Refused(
            "mission-status-diagnostic-observations-mismatch",
        ));
    }
    Ok(agents
        .iter()
        .zip(diagnostics)
        .map(
            |(agent, diagnostic)| crate::mission_status_tasks::AgentObservation {
                agent,
                observed_state: &agent.status,
                fallback_diagnostic: diagnostic,
                environment_recovery: environment_recovery(agent, diagnostic),
            },
        )
        .collect())
}
/// Observe the real current review before building result facts. Complete Work,
/// settings and full-history/view eligibility must already be checked by Root.
/// Semantic causes stay in the supplied sidecar; facts causes are also retained.
/// No new authority, defaults, launch or mutation follows. Unknown read/close
/// errors refuse, and caller still verifies its enclosing material snapshot.
pub(super) fn observed_result_inputs(root:&Path,work:&crate::Work,task:&crate::Task,agent:Option<&Agent>,settings:&crate::EngineSettingsDocument,causes:&mut Vec<E>,mut facts:super::TaskFreshnessEvidence)->Result<crate::mission_status_tasks::ResultObservation,E> {
    causes.append(&mut facts.causes);
    let review=match agent {
        Some(agent) if task.status!="accepted"=>Some(super::mission_status_review::observe(root,work,task,agent,settings,causes)?),
        Some(_)=>Some(crate::mission_status_tasks::ReviewObservation::NotApplicable),
        None=>None,
    };
    result_inputs_with_review(root,task,agent,facts.freshness,facts.gate,review)
}

/// Test-only constructor adapter: unsupported non-passed opinions still refuse
/// without a real producer. Supplied facts establish no actual Root evidence.
#[cfg(test)]
pub(super) fn result_inputs(root:&Path,task:&crate::Task,agent:Option<&Agent>,freshness:crate::mission_status_tasks::FreshnessObservation,gate:crate::mission_status_tasks::GateObservation)->Result<crate::mission_status_tasks::ResultObservation,E> {
    result_inputs_with_review(root,task,agent,freshness,gate,None)
}

/// Observe only facts consumed by the applicable ordinary terminal result branch.
/// Accepted results consume the supplied current gate/freshness before any
/// report/review/readiness projection. This producer never accepts a task.
/// Active and failed/interrupted results use the real docs walk; early reports
/// remain unverified. Completed/submitted results
/// consume the retained submitted report path and do not scan docs or attribute a
/// fresh report; reports/attribution/incomplete remain explicitly NotObserved.
/// Callers must run supported() on full history: ordinary Work or the bounded
/// nonmanaged planned active, failed/interrupted or submitted branches admitted
/// there. Display-window absence of an Agent cannot establish eligibility;
/// supported_view() must also establish the selected branch's required facts.
/// Current review is absent or passed before NotApplicable is supplied, matching
/// Go’s passed-opinion fallthrough. Previous opinions still refuse; non-passed
/// opinions require the actual supplied projection.
/// This is no verdict check. Accepted freshness separately checks native originals;
/// no IO or permission is inferred from skipping the review presentation.
fn result_inputs_with_review(
    root: &Path,
    task: &crate::Task,
    agent: Option<&Agent>,
    freshness: crate::mission_status_tasks::FreshnessObservation,
    gate: crate::mission_status_tasks::GateObservation,
    observed_review: Option<crate::mission_status_tasks::ReviewObservation>,
) -> Result<crate::mission_status_tasks::ResultObservation, E> {
    use crate::mission_status_tasks::*;
    if let Some(agent) = agent {
        if (review_requires_projection(task) && observed_review.is_none())
            || task
                .previous_reviews
                .as_ref()
                .is_some_and(|rows| !rows.is_empty())
        {
            return Err(E::Refused("mission-status-terminal-review-unported"));
        }
        if agent.status == "completed" && !matches!(task.status.as_str(), "submitted" | "accepted")
            && !(task.status=="blocked" && matches!(&observed_review,Some(ReviewObservation::Presented(_)))) {
            return Err(E::Refused("mission-status-completed-unsubmitted-unported"));
        }
    }
    Ok(ResultObservation {
        freshness,
        gate,
        reports: if agent.is_some_and(|agent| {
            agent.is_active() || matches!(agent.status.as_str(), "failed" | "interrupted")
        }) {
            ReportListing::Observed(task_reports(root, &task.id)?)
        } else {
            ReportListing::NotObserved
        },
        attribution: ReportAttribution::NotObserved,
        incomplete_report: IncompleteReport::NotObserved,
        review: observed_review.unwrap_or_else(|| if agent.is_some() {
            ReviewObservation::NotApplicable
        } else {
            ReviewObservation::NotObserved
        }),
    })
}
/// Observe selected-provider active cooldowns at the caller's observation time.
/// Uses the same strict descriptor reader as admission, with provider names sorted
/// and deduplicated as Go missionProviderCooldowns. Expiry and clearance exclude
/// a record; absence is an empty observation, never permission to launch. Invalid
/// records/read/close failures keep their typed launch cause. No writes/provider
/// execution occur. Repeated reads detect observed changes, not an atomic snapshot.
pub(super) fn active_cooldowns(
    root: &Path,
    work: &crate::Work,
    at: i64,
) -> Result<Vec<crate::ProviderCooldown>, E> {
    let mut names = std::collections::BTreeSet::new();
    if let Some(planning) = work.planning.as_deref() {
        if !planning.provider.is_empty() {
            names.insert(planning.provider.as_str());
        }
        if let Some(reviewer) = planning.reviewer.as_deref() {
            if !reviewer.provider.is_empty() {
                names.insert(reviewer.provider.as_str());
            }
        }
    }
    if let Some(profile) = work.launch_profile.as_deref() {
        if !profile.provider.is_empty() {
            names.insert(profile.provider.as_str());
        }
    }
    for task in work.tasks.as_deref().unwrap_or(&[]) {
        if !matches!(task.status.as_str(), "accepted" | "abandoned" | "waived") {
            if let Some(profile) = task.launch_profile.as_deref() {
                if !profile.provider.is_empty() {
                    names.insert(profile.provider.as_str());
                }
            }
        }
    }
    let mut active = Vec::new();
    for name in names {
        let observed = crate::agent_launch::with_provider_cooldown(root, name, Ok)
            .map_err(|e| E::Launch(Box::new(e)))?;
        if let Some(value) = observed {
            if value.cleared_at.is_empty() && (value.resets_at == 0 || at < value.resets_at) {
                active.push(value);
            }
        }
    }
    Ok(active)
}

/// Port resultPresentation's ordered branches without IO. The Agent must be the
/// latest applicable row selected from the supplied history. Freshness, review,
/// reports and gate state are actual observations, never authoritative by being
/// supplied here. Missing consumed evidence refuses; serialization grants nothing.
pub fn result_presentation(
    task: &Task,
    agent: Option<&AgentObservation<'_>>,
    facts: &ResultObservation,
) -> Result<ResultPresentation, E> {
    if agent.is_some_and(|a| a.agent.task_id != task.id || a.observed_state.is_empty()) {
        return Err(E::Refused("mission-status-agent-observation-mismatch"));
    }
    let mut p = ResultPresentation {
        process_label: String::new(),
        report_label: String::new(),
        validation_label: String::new(),
        state: "not_started".into(),
        label: "Pas encore commencé".into(),
        process_state: "not_started".into(),
        report_state: "absent".into(),
        validation_state: "not_started".into(),
        reason: "Aucune tentative terminée ni validation actuelle n’est établie.".into(),
        next_step: "Lancer la tâche lorsque ses conditions sont réunies.".into(),
        attempt_id: String::new(),
        report_id: String::new(),
        gate_id: String::new(),
        receipt_id: String::new(),
    };
    if let Some(a) = agent {
        p.attempt_id = if a.agent.attempt_id.is_empty() {
            a.agent.id.clone()
        } else {
            a.agent.attempt_id.clone()
        };
        p.process_state = a.observed_state.into();
    }
    if let Some(r) = task
        .revalidation
        .as_deref()
        .map(|r| crate::text::trim_space(&r.report))
        .filter(|r| !r.is_empty())
    {
        p.report_id = r.into();
    } else if let Some(at) = task.next.rfind("handoff : ") {
        p.report_id = crate::text::trim_space(&task.next[at + "handoff : ".len()..]).into();
    }
    if !p.report_id.is_empty() || matches!(task.status.as_str(), "submitted" | "accepted") {
        p.report_state = "submitted".into();
    }
    if let Some(g) = &task.gate {
        p.gate_id = if crate::text::trim_space(&g.name).is_empty() {
            g.evaluation.scope_id.clone()
        } else {
            crate::text::trim_space(&g.name).into()
        };
    }
    if let Some(v) = &task.automatic_validation {
        p.receipt_id = v.receipt.clone();
    }
    if task.status == "accepted" {
        match &facts.freshness {
            FreshnessObservation::NotObserved => {
                return Err(E::Refused("mission-status-freshness-unobserved"))
            }
            FreshnessObservation::Current => {
                if !matches!(facts.gate, GateObservation::Current) {
                    return Err(E::Refused("mission-status-current-gate-unobserved"));
                }
                p.state = "validated".into();
                p.label = "Validé".into();
                p.validation_state = "fresh".into();
                p.reason = "Gate delivery et preuves actuelles ; la tâche est acceptée.".into();
                p.next_step = "Consulter le résultat ; revalider si une preuve change.".into();
            }
            FreshnessObservation::NotCurrent { blockers } => {
                p.state = "validation_stale".into();
                p.label = "Validation à refaire".into();
                p.validation_state = "stale".into();
                p.reason = blockers.first().cloned().unwrap_or_else(|| {
                    "L’acceptation historique n’est plus fondée sur des preuves actuelles.".into()
                });
                p.next_step="Rouvrir la tâche, renouveler les preuves et enregistrer une nouvelle revue de validation.".into();
            }
        }
        return Ok(finish_result(p));
    }
    if task.status == "waived" {
        p.state = "waived".into();
        p.label = "Dérogation — non validé".into();
        p.validation_state = "waived".into();
        p.reason =
            "Une décision explicite permet la suite sans transformer les contrôles en PASS.".into();
        p.next_step = "Consulter le motif et la portée de la dérogation.".into();
        return Ok(finish_result(p));
    }
    if task.status == "abandoned" {
        p.state = "abandoned".into();
        p.label = "Abandonné — non validé".into();
        p.validation_state = "abandoned".into();
        p.reason = "La tâche est abandonnée ; aucun résultat validé n’est revendiqué.".into();
        p.next_step = "Consulter la décision d’abandon.".into();
        return Ok(finish_result(p));
    }
    if let Some(a) = agent {
        if a.agent.is_active() {
            if p.process_state == "running" {
                p.state = "in_progress".into();
                p.label = "En cours".into();
                p.reason =
                    "Le processus est actif ; aucun résultat final n’est encore affirmé.".into();
                p.next_step = "Voir la session ou attendre la fin de la tentative.".into();
            } else {
                p.state = "execution_unconfirmed".into();
                p.label = "État à vérifier".into();
                p.reason =
                    "La tentative est enregistrée, mais son exécution n’est pas confirmée.".into();
                p.next_step = "Réconcilier la tentative avant toute relance.".into();
            }
            if let Some(report) = first_report(&facts.reports)? {
                p.report_state = "early_unverified".into();
                p.report_id = report.into();
                p.reason
                    .push_str(" Un rapport existe déjà, mais il peut être précoce ou incomplet.");
            }
            p.validation_state = "pending_process".into();
            return Ok(finish_result(p));
        }
        if matches!(a.agent.status.as_str(), "failed" | "interrupted") {
            if apply_review(&mut p, &facts.review)? {
                return Ok(finish_result(p));
            }
            p.state = "stopped_early".into();
            p.label = "Arrêté avant la fin".into();
            p.validation_state = "not_validated".into();
            p.reason = crate::text::trim_space(&a.agent.activity).into();
            if p.reason.is_empty() {
                p.reason = if a.agent.status == "failed" {
                    "Le processus s’est terminé en échec."
                } else {
                    "Le processus a été interrompu."
                }
                .into();
            }
            if let Some(report) = first_report(&facts.reports)? {
                if p.report_state == "absent" {
                    p.report_state = "partial_possible".into();
                    p.report_id = report.into();
                    p.reason.push_str(" Un rapport existe, mais l’arrêt ne permet pas d’en affirmer la complétude.");
                }
            }
            p.next_step="Examiner le motif, les traces et le rapport éventuel avant de décider d’une reprise.".into();
            return Ok(finish_result(p));
        }
        if a.agent.status == "completed"
            && task.status == "blocked"
            && task.blocker.starts_with("Livraison incomplète :")
        {
            p.state = "delivery_incomplete".into();
            p.label = "Résultat à compléter".into();
            p.validation_state = "not_validated".into();
            p.reason = task.blocker.clone();
            p.next_step="Le responsable doit examiner les critères manquants et préparer une correction dans les limites autorisées.".into();
            match &facts.incomplete_report {
                IncompleteReport::NotObserved => {
                    return Err(E::Refused("mission-status-incomplete-report-unobserved"))
                }
                IncompleteReport::Observed(Some(report)) if !report.is_empty() => {
                    p.report_id = report.clone();
                    p.report_state = "partial_possible".into();
                }
                IncompleteReport::Observed(_) => {}
            }
            return Ok(finish_result(p));
        }
        if a.agent.status == "completed" && apply_review(&mut p, &facts.review)? {
            return Ok(finish_result(p));
        }
    }
    if task.status == "submitted" {
        if agent.is_none() {
            p.process_state = "unknown".into();
        }
        p.validation_state = "missing_gate".into();
        p.state = "result_to_review".into();
        p.label = "Résultat à vérifier".into();
        p.reason="Un rapport est soumis ; sa complétude et les critères restent à examiner. Aucune revue de validation n’est enregistrée.".into();
        p.next_step =
            "Lire le rapport, vérifier les critères et enregistrer la revue de validation.".into();
        if task.gate.is_some() {
            let valid = match facts.gate {
                GateObservation::Current => true,
                GateObservation::NotCurrent => false,
                _ => return Err(E::Refused("mission-status-gate-unobserved")),
            };
            let blockers = blockers(&facts.freshness)?;
            if !valid || !blockers.is_empty() {
                p.state = "validation_withheld".into();
                p.label = "Validation retenue".into();
                p.validation_state = "failed_or_stale".into();
                p.reason = blockers.first().cloned().unwrap_or_else(|| {
                    "La revue de validation n’autorise pas une validation actuelle.".into()
                });
                p.next_step="Corriger ou renouveler les preuves, puis enregistrer une revue de validation actuelle.".into();
            } else {
                p.validation_state = "ready_for_decision".into();
                p.reason="La revue de validation et ses preuves sont actuelles ; la décision d’acceptation reste à enregistrer.".into();
                p.next_step =
                    "Examiner le rapport puis enregistrer la décision de validation.".into();
            }
        }
        return Ok(finish_result(p));
    }
    if agent.is_some_and(|a| a.agent.status == "completed") {
        p.process_state = "completed".into();
        p.validation_state = "not_started".into();
        match &facts.attribution {
            ReportAttribution::NotObserved => {
                return Err(E::Refused("mission-status-report-attribution-unobserved"))
            }
            ReportAttribution::Attributed(report) => {
                if report.is_empty() {
                    return Err(E::Refused("mission-status-report-empty"));
                }
                p.state = "result_to_review".into();
                p.label = "Résultat à vérifier".into();
                p.report_state = "attributable_unsubmitted".into();
                p.report_id = report.clone();
                p.reason="Le processus est terminé et un rapport lui est attribuable ; sa complétude n’est pas démontrée.".into();
                p.next_step="Soumettre le rapport, vérifier les critères et enregistrer la revue de validation.".into();
            }
            ReportAttribution::Unavailable { reason } => {
                p.state = "completed_unproven".into();
                p.label = "Terminé, résultat non démontré".into();
                p.report_state = "absent_or_unattributed".into();
                p.reason = format!("Le processus est terminé normalement, mais {reason}.");
                if let Some(a) = agent {
                    if !crate::text::trim_space(&a.agent.relay).is_empty() {
                        p.reason = a.agent.relay.clone();
                    }
                }
                p.next_step =
                    "Désigner ou compléter un rapport avant toute validation ou relance.".into();
            }
        }
        return Ok(finish_result(p));
    }
    if task.gate.is_some() {
        match facts.gate {
            GateObservation::Current => {}
            GateObservation::NotCurrent => p.validation_state = "failed_or_stale".into(),
            _ => return Err(E::Refused("mission-status-gate-unobserved")),
        }
    }
    Ok(finish_result(p))
}

/// Project the latest applicable attempt and model/cost attribution exactly from
/// retained rows. Absence uses Task.attempts history, never current model settings.
fn attempt(task: &Task, agents: &[AgentObservation<'_>]) -> MissionAttemptProjection {
    let mut p = MissionAttemptProjection {
        agent_id: String::new(),
        attempt_id: String::new(),
        role: String::new(),
        process_state: String::new(),
        activity: String::new(),
        validation: task.status.clone(),
        requested_model: String::new(),
        observed_model: String::new(),
        cost_state: "unknown".into(),
    };
    if let Some(a) = agents.iter().find(|a| a.agent.task_id == task.id) {
        p.agent_id = a.agent.id.clone();
        p.attempt_id = a.agent.attempt_id.clone();
        p.role = a.agent.role.clone();
        p.process_state = a.agent.status.clone();
        p.activity = a.agent.activity.clone();
        if let Some(model) = &a.agent.model_route {
            p.requested_model = model.model.clone();
        }
        if let Some(model) = &a.agent.reported_model {
            p.observed_model = model.model.clone();
        }
        if a.agent
            .usage
            .as_ref()
            .is_some_and(|u| u.provider_reported_cost_usd.is_some())
        {
            p.cost_state = "reported".into();
        }
    } else if let Some(last) = task.attempts.as_ref().and_then(|a| a.last()) {
        p.attempt_id = last.id.clone();
    }
    p
}
/// Count transitive dependants with a visited set as Go descendantCount does.
/// This is a display impact, not graph validation, ordering or scheduling.
fn impact(work: &Work, id: &str) -> Result<i64, E> {
    let mut seen = BTreeSet::from([id]);
    loop {
        let old = seen.len();
        for task in work.tasks.iter().flatten() {
            if task
                .depends
                .iter()
                .flatten()
                .any(|d| seen.contains(d.as_str()))
            {
                seen.insert(task.id.as_str());
            }
        }
        if old == seen.len() {
            break;
        }
    }
    i64::try_from(seen.len() - 1).map_err(|_| E::Refused("mission-status-count-representation"))
}
/// Validate exact dependency coordinates before reading freshness; every real
/// dependency is observed even for accepted tasks because taskWaits consumes it.
fn dependencies<'a>(
    task: &Task,
    facts: &'a [DependencyObservation],
) -> Result<&'a [DependencyObservation], E> {
    let ids = task.depends.as_deref().unwrap_or(&[]);
    if ids.len() != facts.len() || ids.iter().zip(facts).any(|(id, f)| id != &f.task_id) {
        return Err(E::Refused("mission-status-dependencies-mismatch"));
    }
    if facts
        .iter()
        .any(|f| matches!(f.freshness, DependencyFreshness::NotObserved))
    {
        return Err(E::Refused("mission-status-dependency-unobserved"));
    }
    Ok(facts)
}
/// Read-only dependency explanations in stored order. Missing/stale facts were
/// classified by the reader; no acceptance or filesystem checks occur here.
fn waits(work: &Work, facts: &[DependencyObservation]) -> Vec<MissionWait> {
    facts.iter().filter(|f|matches!(f.freshness,DependencyFreshness::NotCurrent)).map(|f|{
        match work.tasks.iter().flatten().find(|t|t.id==f.task_id){
            None=>MissionWait {task:f.task_id.clone(),title:f.task_id.clone(),state:"missing".into(),reason:"Prérequis absent du plan.".into()},
            Some(parent)=>MissionWait {task:f.task_id.clone(),title:parent.title.clone(),state:if parent.status=="accepted"{"stale".into()}else{parent.status.clone()},reason:if parent.status=="accepted"{"Validation ancienne : les preuves actuelles doivent être vérifiées à nouveau."}else{"Ce résultat n’est pas encore validé."}.into()}
        }
    }).collect()
}
/// Pure recoveryPreviewFor text only. No dialogue/receipt inspection or recovery
/// consent is implied; optional advanced preview fields remain nil as Go does.
fn recovery(work: &Work, task: &Task, agent: Option<&AgentObservation<'_>>) -> RecoveryPreview {
    let mut p=RecoveryPreview {dialogue:None,work:work.id.clone(),task:task.id.clone(),revision:work.revision,agent:String::new(),kept:Some(vec!["Historique des tentatives, rapports et avis précédents.".into(),"Critères et dépendances du plan.".into()]),redone:Some(vec!["Vérification des conditions de lancement et du candidat courant.".into(),"Contrôles et revue requis pour la nouvelle tentative.".into()]),correction:task.next.clone(),criteria:Some(task.criteria.as_deref().unwrap_or(&[]).to_vec()),limits:"Aucun budget ni plafond n’est augmenté. Ce résumé n’autorise aucun départ et ne valide aucun résultat.".into(),since_refusal:None,evidence:None,remaining:None,refusal_note:String::new(),evidence_note:String::new()};
    if let Some(a) = agent {
        p.agent = a.agent.id.clone();
        if let Some(kept) = &mut p.kept {
            kept.push("Fournisseur et espace de la tentative sélectionnée, sous réserve des règles du lancement.".into());
        }
        if let Some(redone) = &mut p.redone {
            redone.push(
                "Une nouvelle tentative d’agent ; l’ancienne reste dans l’historique.".into(),
            );
        }
    }
    if let Some(r) = &task.corrective_recovery {
        if !r.instruction.is_empty() {
            p.correction = r.instruction.clone();
        }
    }
    if work
        .planning
        .as_ref()
        .is_some_and(|p| p.repository.is_some())
    {
        if let Some(kept) = &mut p.kept {
            kept.push("Résultat précédent réutilisable seulement après contrôle du dossier de reprise ; ancienne copie conservée.".into());
        }
    }
    p
}
/// Apply only a producer-observed dispatch disposition. Structural eligibility,
/// slot/pause checks and recovery classification are not reimplemented or guessed.
fn disposition(observed: &ReadinessObservation) -> Result<(&'static str, &str), E> {
    match observed {
        ReadinessObservation::NotObserved => Err(E::Refused("mission-status-readiness-unobserved")),
        ReadinessObservation::Observed {
            disposition,
            reason,
            ..
        } => Ok((
            match disposition {
                DispatchDisposition::Ready => "ready",
                DispatchDisposition::Waiting => "waiting",
                DispatchDisposition::Manual => "manual",
                DispatchDisposition::Intervention => "intervention",
            },
            reason,
        )),
    }
}
/// Configure default task-state branches from exact dependency/readiness inputs.
/// The repeated dispatch observation used for corrective retries grants nothing.
fn initial_task_state(
    x: &mut MissionTask,
    work: &Work,
    task: &Task,
    context: &MissionStatus,
    facts: &TaskObservation,
) -> Result<(), E> {
    x.action = "inspect".into();
    x.label = "Comprendre et résoudre".into();
    x.reason = task.blocker.clone();
    for dependency in &facts.dependencies {
        if matches!(dependency.freshness, DependencyFreshness::NotCurrent) {
            x.state = "waiting".into();
            x.target = dependency.task_id.clone();
            x.reason = work
                .tasks
                .iter()
                .flatten()
                .find(|t| t.id == dependency.task_id)
                .map(|p| format!("Attend {}", p.title))
                .unwrap_or_else(|| format!("Attend la validation de {}", dependency.task_id));
            x.label = "Examiner le prérequis".into();
            break;
        }
    }
    if x.state == "waiting" {
        return Ok(());
    }
    let (state, reason) = disposition(&facts.readiness)?;
    x.state = state.into();
    x.reason = reason.into();
    let ReadinessObservation::Observed {
        profile,
        assistance,
        workspace,
        ..
    } = &facts.readiness
    else {
        return Err(E::Refused("mission-status-readiness-unobserved"));
    };
    if matches!(profile, ProfileObservation::NotObserved) {
        return Err(E::Refused("mission-status-profile-unobserved"));
    }
    if x.state == "ready" {
        match assistance {
            AssistObservation::NotObserved => {
                return Err(E::Refused("mission-status-assist-unobserved"))
            }
            AssistObservation::Held { reason } => {
                x.state = "intervention".into();
                x.reason = reason.clone();
            }
            AssistObservation::Allowed => {
                if matches!(profile, ProfileObservation::Missing) {
                    return Err(E::Refused("mission-status-ready-profile-missing"));
                }
                match workspace {
                    WorkspaceObservation::NotObserved => {
                        return Err(E::Refused("mission-status-workspace-unobserved"))
                    }
                    WorkspaceObservation::Occupied => {
                        x.state = "waiting".into();
                        x.reason =
                            "Espace occupé par une tentative ; reprise après sa libération".into();
                    }
                    WorkspaceObservation::Free => {}
                }
                if x.state == "ready" {
                    x.action = "start".into();
                    x.label = "Lancer cette tâche".into();
                    if !context.authorized {
                        x.state = "manual".into();
                        x.reason =
                            "Lancement manuel possible ; mission continue non autorisée".into();
                    } else if !context.enabled {
                        x.state = "waiting".into();
                        x.action = "supervise".into();
                        x.label = "Voir comment reprendre".into();
                        x.reason = if context.supervision.state == "error" {
                            "Conducteur présent, mais sa dernière vérification est en erreur"
                        } else {
                            "Autorisation enregistrée ; aucun conducteur actif observé"
                        }
                        .into();
                    }
                }
            }
        }
    }
    if matches!(profile, ProfileObservation::Missing) && x.state == "intervention" {
        x.state = "configure".into();
        x.reason = "Configuration initiale à préparer avant le premier départ".into();
        x.action = "configure".into();
        x.label = "Préparer le lancement".into();
    }
    if task.launch_held {
        x.state = "intervention".into();
        x.action = "prepare".into();
        x.label = "Autoriser le plan dans Préparer".into();
    }
    Ok(())
}

/// Project all tasks in Work order, returning a complete new value or an error.
/// Inputs must cover every task exactly once and every dependency in stored order;
/// Agent facts retain the producer's latest-first history window. No partial
/// projection is returned on missing proof, coordinate mismatch or numeric error.
/// Counts follow Go's result branches before prepared/recovery task overrides.
/// Existing insight functions produce explanations/actions; no fresh grant follows.
/// Current/managed review and readiness classifiers remain mandatory external
/// producers: this function does not implement or silently approximate them.
pub fn project_tasks(
    work: &Work,
    context: &MissionStatus,
    observations: &[TaskObservation],
    agents: &[AgentObservation<'_>],
) -> Result<TaskProjection, E> {
    let tasks = work.tasks.as_deref().unwrap_or(&[]);
    if tasks.len() != observations.len()
        || tasks
            .iter()
            .zip(observations)
            .any(|(t, o)| t.id != o.task_id)
    {
        return Err(E::Refused("mission-status-task-observations-mismatch"));
    }
    let mut context = context.clone();
    context.validated = 0;
    context.review = 0;
    context.running = 0;
    context.total = i64::try_from(tasks.len())
        .map_err(|_| E::Refused("mission-status-count-representation"))?;
    let insight_agents: Vec<_> = agents
        .iter()
        .map(|a| MissionAgentInsight {
            task_id: &a.agent.task_id,
            status: &a.agent.status,
            active: a.agent.is_active(),
            observed_state: a.observed_state,
            fallback_diagnostic_limit_reached: a.fallback_diagnostic.limit_reached,
            environment_recovery: a.environment_recovery,
        })
        .collect();
    let mut result = TaskProjection {
        tasks: Vec::with_capacity(tasks.len()),
        validated: 0,
        review: 0,
        running: 0,
    };
    for (task, facts) in tasks.iter().zip(observations) {
        let deps = dependencies(task, &facts.dependencies)?;
        let agent = agents.iter().find(|a| a.agent.task_id == task.id);
        let result_facts = result_presentation(task, agent, &facts.result)?;
        let mut x = MissionTask {
            waits: Some(waits(work, deps)),
            recovery_preview: recovery(work, task, agent),
            primary: MissionPrimaryAction {
                kind: String::new(),
                label: String::new(),
                effect: String::new(),
                task: String::new(),
                tone: String::new(),
            },
            attempts_used: i64::try_from(task.attempts.as_ref().map_or(0, Vec::len))
                .map_err(|_| E::Refused("mission-status-count-representation"))?,
            attempts_allowed: task.plan_max_attempts,
            attempt_limit_reached: false,
            id: task.id.clone(),
            title: task.title.clone(),
            state: task.status.clone(),
            reason: String::new(),
            action: String::new(),
            label: String::new(),
            target: task.id.clone(),
            impact: impact(work, &task.id)?,
            deliverable: task.deliverable.clone(),
            validation_mode: task
                .validation_policy
                .as_ref()
                .map_or_else(|| "human".into(), |v| v.mode.clone()),
            validation_receipt: task
                .automatic_validation
                .as_ref()
                .map_or_else(String::new, |v| v.receipt.clone()),
            result: result_facts,
            understanding: MissionUnderstanding {
                what: String::new(),
                next_step: String::new(),
                actor: String::new(),
                actor_kind: String::new(),
                situation: String::new(),
            },
            diagnostic: None,
            attempt: attempt(task, agents),
        };
        let count = |field: &mut i64| -> Result<(), E> {
            *field = field
                .checked_add(1)
                .ok_or(E::Refused("mission-status-count-representation"))?;
            Ok(())
        };
        match x.result.state.as_str() {
            "validated" => {
                count(&mut result.validated)?;
                x.state = "validated".into();
                x.reason = x.result.reason.clone();
                x.action = "report".into();
                x.label = "Voir le résultat".into();
            }
            "validation_stale" => {
                x.state = "intervention".into();
                x.reason = x.result.reason.clone();
                x.action = "reopen".into();
                x.label = "Revalider les preuves".into();
            }
            "waived" | "abandoned" => {
                x.state = x.result.state.clone();
                x.reason = x.result.reason.clone();
                x.action = "inspect".into();
                x.label = if x.state == "waived" {
                    "Voir la dérogation"
                } else {
                    "Voir la décision"
                }
                .into();
            }
            "review_blocked" | "delivery_incomplete" => {
                x.state = "intervention".into();
                x.reason = x.result.reason.clone();
                x.action = "inspect".into();
                x.label = "Examiner le résultat".into();
            }
            "review_in_progress" | "review_awaiting_publication" => {
                count(&mut result.review)?;
                x.state = "review".into();
                x.reason = x.result.reason.clone();
                x.action = "report".into();
                x.label = "Examiner le résultat".into();
            }
            "result_to_review" | "validation_withheld" => {
                count(&mut result.review)?;
                x.state = "review".into();
                x.reason = x.result.reason.clone();
                x.action = "report".into();
                x.label = "Examiner le résultat".into();
                if x.validation_mode == "automatic" && x.result.validation_state == "missing_gate" {
                    x.reason="Contrôles automatiques préautorisés retenus ; vérifier pause, autorisation et preuves".into();
                }
                if x.result.validation_state == "ready_for_decision" {
                    x.action = "accepted".into();
                    x.label = "Valider après revue".into();
                }
            }
            "in_progress" | "execution_unconfirmed" => {
                count(&mut result.running)?;
                x.state = "running".into();
                x.reason = x.result.reason.clone();
                x.action = "inspect".into();
                x.label = "Suivre l’agent".into();
            }
            "stopped_early" | "completed_unproven" => {
                x.state = "intervention".into();
                x.reason = x.result.reason.clone();
                x.action = "inspect".into();
                x.label = "Examiner avant reprise".into();
            }
            _ => initial_task_state(&mut x, work, task, &context, facts)?,
        }
        match &facts.prepared_launch {
            PreparedLaunchObservation::NotObserved => {
                return Err(E::Refused("mission-status-prepared-launch-unobserved"))
            }
            PreparedLaunchObservation::Absent => {}
            PreparedLaunchObservation::Present { reason } => {
                x.state = "intervention".into();
                x.action = "inspect".into();
                x.label = "Reprendre le lancement préparé".into();
                x.reason = if reason.is_empty() {
                    "La copie et les réglages sont conservés. Confirmez la reprise du lancement depuis cette tâche.".into()
                } else {
                    reason.clone()
                };
            }
        }
        if task.status == "blocked"
            && task.plan_max_attempts > 0
            && x.attempts_used >= task.plan_max_attempts
            && x.state != "review"
        {
            x.attempt_limit_reached = match facts.execution {
                ExecutionObservation::NotObserved => {
                    return Err(E::Refused("mission-status-execution-unobserved"))
                }
                ExecutionObservation::Observed => false,
                ExecutionObservation::NotPresent => true,
            };
        }
        if task
            .independent_review
            .as_ref()
            .is_some_and(|r| r.state == "running")
            || agents
                .iter()
                .any(|a| a.agent.task_id == task.id && a.agent.is_active())
        {
            x.attempt_limit_reached = false;
        }
        if x.attempt_limit_reached {
            x.state = "intervention".into();
            x.action = "recovery".into();
            x.label = "Examiner les tentatives et les refus".into();
            if let Some(r) = &task.independent_review {
                if r.state == "changes_requested" {
                    x.reason = r.reason.clone();
                }
            }
        }
        if task.corrective_recovery.is_some()
            && task.planning_retry
            && task.status == "blocked"
            && x.attempts_used < task.plan_max_attempts
        {
            let (state, reason) = disposition(&facts.readiness)?;
            x.state = state.into();
            x.reason = format!("Essai correctif autorisé. {reason}");
            x.action = "supervise".into();
            x.label = "Suivre la reprise".into();
        }
        context.validated = result.validated;
        context.review = result.review;
        context.running = result.running;
        x.understanding = task_understanding(&x, &context, &insight_agents);
        if let Some(a) = agent {
            if a.fallback_diagnostic
                .items
                .as_ref()
                .is_some_and(|v| !v.is_empty())
            {
                x.diagnostic = Some(Box::new(a.fallback_diagnostic.clone()));
            }
        }
        x.primary = mission_task_primary(&x);
        result.tasks.push(x);
    }
    Ok(result)
}

//! Go result_presentation.go currentReviewPresentation ordering. Recorded review

/// Require the actual latest attempt and producer, without normalization or a
/// chronological guess from opaque identifiers. Passed deliberately falls
/// through as Go; this pure predicate verifies no evidence or authority.
pub(super) fn applies(
    planning: bool,
    latest_attempt: Option<&str>,
    agent_id: &str,
    agent_attempt: &str,
    review: &IndependentReview,
) -> bool {
    planning
        && review.state != "passed"
        && latest_attempt == Some(review.attempt.as_str())

/// Match the historical presentation vocabulary; unknown is not a passed opinion.

    p.validation_state = "failed_or_stale".into();

/// Check the contained report bytes against their recorded digest. Known stale
/// errors are retained in the sidecar and return false; unknown IO/cleanup errors
/// refuse the whole read. No unreadable file becomes absence or an empty success.
fn report_current(
    root: &Path,
    review: &IndependentReview,
    p: &mut ReviewPresentation,
    causes: &mut Vec<E>,
) -> Result<bool, E> {
    match artifact_hash(root, &review.report) {
        Ok((_, hash)) if hash == review.sha256 => Ok(true),
        Ok(_) => {
            causes.push(E::Refused("mission-status-review-report-stale"));
            stale_report(p);
            Ok(false)
        }
        Err(error) if semantic_artifact(&error) => {
            causes.push(E::Gate(crate::GateContractError::Evaluation(error)));
            stale_report(p);
            Ok(false)
        }
        Err(error) => Err(E::Gate(crate::GateContractError::Evaluation(error))),
    }
}

/// Check every declared review artifact. A digest mismatch or known semantic
/// failure returns false with its typed sidecar; unknown IO/close errors refuse.
/// Missing map is exactly the stored absence, not a fabricated evidence list.
fn artifacts_current(
    root: &Path,
    review: &IndependentReview,
    causes: &mut Vec<E>,
) -> Result<bool, E> {
    for (name, expected) in review.report_artifacts.iter().flatten() {
        match artifact_hash(root, name) {
            Ok((_, actual)) if &actual == expected => {}
            Ok(_) => {
                causes.push(E::Refused("mission-status-review-artifact-stale"));
                return Ok(false);
            }
            Err(error) if semantic_artifact(&error) => {
                causes.push(E::Gate(crate::GateContractError::Evaluation(error)));
                return Ok(false);
            }
            Err(error) => return Err(E::Gate(crate::GateContractError::Evaluation(error))),
        }
    }
    Ok(true)
}

/// Distinguish stale control evidence from unknown IO/configuration/codec errors.
/// Quality policy absence is a missing precondition, never an unfavorable verdict.
fn semantic_validation(error: &crate::ValidationEvidenceError) -> bool {
    match error {
        crate::ValidationEvidenceError::Refused(code) => matches!(
            *code,
            "current-validation-required"
                | "successful-controls-required"
                | "complete-receipt-required"
                | "control-not-demonstrated"
                | "control-output-not-authorized"
                | "control-output-incoherent"
                | "validation-artifact-stale"
        ),
        crate::ValidationEvidenceError::Gate(error) => semantic_gate(error),
        crate::ValidationEvidenceError::Artifact(error) => semantic_artifact(error),
        crate::ValidationEvidenceError::Encode(_) => false,
    }
}

/// Match the known Go control-refusal vocabulary for the static receipt cases.
/// Other typed causes retain their safe Rust diagnostic; this formatting neither
/// suppresses the original cause nor establishes verbatim parity for all errors.
fn validation_reason(error:&crate::ValidationEvidenceError)->String {
    match error {
        crate::ValidationEvidenceError::Refused("current-validation-required")=>"contrôles moteur courants requis avant la revue".into(),
        crate::ValidationEvidenceError::Refused("successful-controls-required")=>"contrôles non réussis".into(),
        crate::ValidationEvidenceError::Refused("complete-receipt-required")=>"reçu moteur incomplet".into(),
        _=>error.to_string(),
    }
}

/// Observe the applicable ordinary review using current contained artifacts and
/// contract, with Go's passed/no-planning/stale-attempt fallthrough. Managed
/// presentation remains refused. Before first required review, current controls
/// use saved validation-output/quality settings; semantic failures are explicit
/// ControlsStale with original causes, unknown failures remain Err. Caller
/// supplies a real canonical Work/settings snapshot under material association,
/// preserves the sidecar even on later errors, and rechecks its outer snapshot.
/// No launch, SQL mutation, retry, new policy, authentication or acceptance occurs.
pub(super) fn observe(
    root: &Path,
    work: &Work,
    task: &Task,
    agent: &Agent,
    settings: &EngineSettingsDocument,
    causes: &mut Vec<E>,
) -> Result<ReviewObservation, E> {
    let Some(planning) = work.planning.as_deref() else {
        return Ok(ReviewObservation::NotApplicable);
    };
    if planning.repository.is_some() {
        return Err(E::Refused("mission-status-managed-review-unported"));
    }
    let latest = task
        .attempts
        .as_ref()
        .and_then(|v| v.last())
        .map(|a| a.id.as_str());
    if task.independent_review.is_none()
        && planning.reviewer_required
        && task.status == "submitted"
        && agent.status == "completed"
        && latest == Some(agent.attempt_id.as_str())
    {
        let output = settings
            .values
            .validation_output_limit()
            .map_err(E::Settings)?;
        let quality = if task.validation_policy.as_deref().is_some_and(|p| {
            p.mode == "automatic" && p.controls.as_ref().is_some_and(|v| !v.is_empty())
        }) {
            Some(settings.quality_policy_snapshot().map_err(E::Settings)?)
        } else {
            None
        };
        match crate::check_validation_evidence(root, task, quality.as_ref(), output) {
            Ok(_) => {}
            Err(error) if semantic_validation(&error) => {
                let reason = validation_reason(&error);
                causes.push(E::ValidationEvidence(error));
                return Ok(ReviewObservation::ControlsStale { reason });
            }
            Err(error) => return Err(E::ValidationEvidence(error)),
        }
    }
    let Some(review) = task.independent_review.as_deref() else {
        return Ok(ReviewObservation::NotApplicable);
    };
    if !applies(true, latest, &agent.id, &agent.attempt_id, review) {
        return Ok(ReviewObservation::NotApplicable);
    }
    let mut presentation = base(review);
    if !report_current(root, review, &mut presentation, causes)? {
        return Ok(ReviewObservation::Presented(presentation));
    }
    let contract = crate::review_contract(task).map_err(E::Encode)?;
    if review.contract != contract || !artifacts_current(root, review, causes)? {
        if review.contract != contract {
            causes.push(E::Refused("mission-status-review-contract-stale"))
        }
        presentation.validation_state = "failed_or_stale".into();
        presentation.label = "Avis périmé".into();
        presentation.reason =
            "Le contrat ou la révision de base a changé depuis cette revue.".into();
        return Ok(ReviewObservation::Presented(presentation));
    }
    apply_state(&mut presentation, &review.state);
    Ok(ReviewObservation::Presented(presentation))
}

//! observations, never authenticated authorship, quality or task acceptance.
use super::{finished, LocalDirectory, RootStore, RootStoreError as E};

/// Result of inspecting one retained worker, including workers from older watch
/// generations. Waiting observes no submission; Submitted is known recorded
/// Work state, not validation/acceptance. No process is started or signalled.
pub enum MissionReportRelayOutcome {
    NotApplicable,
    Waiting { reason: &'static str },
    Submitted { work: String, revision: i64, agent_id: String, report: String },
}
/// One contained, nonempty report observed after applying the saved clock margin.
#[derive(Debug, PartialEq, Eq)]
struct Report { path: String, sha256: String }
/// Keep original file/clock/codec errors, including nested cleanup failures.
fn caused(code: &'static str, error: impl Error + Send + Sync + 'static) -> E {
    E::Mutation(WorkEngineError::Mission(Box::new(crate::MissionSupervisionError::Cause {
        code, cause: Box::new(error),
    })))
}
/// Select an inclusive freshness window without date subtraction overflow. Inputs
/// are observed nanoseconds and an explicitly validated nonnegative i64 setting;
/// constants only convert seconds to nanoseconds. Empty/stale/ambiguous remain
/// distinct; this pure helper establishes no filesystem containment or identity.
fn select(times: &[(String, u64, i128)], started: i128, skew: i64) -> Result<Result<&str, &'static str>, E> {
    if skew < 0 { return Err(E::Refused("mission-report-clock-skew-invalid")); }
    let since = started - i128::from(skew) * 1_000_000_000;
    let mut found = None;
    let mut stale = false;
    for (path, size, modified) in times {
        if *size == 0 { continue; }
        if *modified < since { stale = true; continue; }
        if found.is_some() { return Ok(Err("mission-report-ambiguous")); }
        found = Some(path.as_str());
    }
    Ok(found.ok_or(if stale { "mission-report-stale" } else { "mission-report-missing" }))
}
/// Discover only in the actual attempt workspace, translate to the canonical
/// project and hash bytes. Shared listing/metadata/hash readers retain unknown
/// IO/path/close causes. Sequential observations are not an atomic file snapshot;
/// the mutation fence repeats selection and task.submit rechecks actual bytes.
fn report(root: &Path, agent: &Agent, skew: i64) -> Result<Result<Report, &'static str>, E> {
    let started = swarm_wire_json::go_rfc3339::parse(&agent.started)
        .map_err(|e| caused("mission-report-start-invalid", e))?.unix_timestamp_nanos();
    let workspace = if agent.workspace.is_empty() { root.to_path_buf() } else {
        crate::canonical_project_root(Path::new(&agent.workspace)).map_err(E::Path)?
    };
    if !workspace.starts_with(root) || !workspace.is_dir() {
        return Err(E::Refused("mission-report-workspace-outside-root"));
    }
    let mut times = Vec::new();
    for name in super::mission_status_reports::task_reports(&workspace, &agent.task_id)? {
        let (_, metadata) = crate::gate_evaluation::artifact_metadata(&workspace, &name)
            .map_err(|e| caused("mission-report-metadata", e))?;
        use std::os::unix::fs::MetadataExt;
        let modified = i128::from(metadata.mtime()) * 1_000_000_000 + i128::from(metadata.mtime_nsec());
        times.push((name, metadata.len(), modified));
    }
    let name = match select(&times, started, skew)? { Ok(name) => name, Err(reason) => return Ok(Err(reason)) };
    let relative = workspace.join(name).strip_prefix(root)
        .map_err(|e| caused("mission-report-relative", e))?.to_path_buf();
    let relative = relative.to_str().ok_or(E::Refused("mission-report-path-not-utf8"))?;
    let (_, sha256) = crate::gate_evaluation::artifact_hash(root, relative)
        .map_err(|e| caused("mission-report-hash", e))?;
    Ok(Ok(Report { path: relative.into(), sha256 }))
}
/// Recheck actual completed worker, original automatic claim and durable terminal
/// receipt against the connected datastore. Require live mission consent/mode,
/// pause/archive guards; no historical label alone authorizes the new mutation.
pub(super) fn producer(db: &Connection, directory: &LocalDirectory, work: &Work, agent_id: &str, settings: &EngineSettingsDocument) -> Result<Agent, E> {
    super::mission_producer::completed(db,directory,work,agent_id,settings,super::mission_producer::OriginRequirement::AutomaticWorker)
}
/// Engine-owned fence: current lease plus original worker proof and exact current
/// selection. Settings are reread under the command's existing reservation; a
/// revision change refuses, never silently adopts a new margin mid-submission.
/// The post-stage check recognizes only its exact own intent (event/revision/
/// report), without treating an older delivered result as a fresh submission.
struct RelayFence<'a> { lease: &'a MissionLease, directory: LocalDirectory, root: &'a Path, agent: &'a str, settings_revision: crate::SettingCount, event_id: &'a str, expected_revision: i64, report: &'a Report }
impl MutationFence for RelayFence<'_> {
    fn stage_receipt(&self, db: &Connection, receipt: MutationFenceReceipt<'_>) -> Result<(), WorkEngineError> { self.lease.stage_receipt(db, receipt) }
    fn check(&self, db: &Connection, work: &str, now: &str) -> Result<(), WorkEngineError> {
        self.lease.check(db, work, now)?;
        let result = (|| {
            let settings = RootStore::read_settings_from(&self.directory)?;
            if settings.revision != self.settings_revision { return Err(E::Refused("mission-report-settings-changed")); }
            let saved = crate::read_recorded_work(db, work).map_err(E::Work)?;
            let row = producer(db, &self.directory, &saved, self.agent, &settings)?;
            if let Some(prior)=super::task_validation::latest_submission(db,&saved,&row.task_id).map_err(|e|E::Launch(Box::new(e)))?.filter(|r|r.agent_id==self.agent){
                // WorkEngine repeats the fence after staging this very intent,
                // before commit. Only that exact canonical own submission is
                // allowed here; preflight already excluded historical delivery.
                if prior.event_id!=self.event_id || prior.expected_revision!=self.expected_revision || prior.report_path!=self.report.path || prior.report_sha256!=self.report.sha256 {
                    return Err(E::Refused("mission-report-already-submitted"));
                }
            }
            let observed = report(self.root, &row, settings.values.report_clock_skew_seconds().map_err(E::Settings)?)?
                .map_err(E::Refused)?;
            if &observed != self.report { return Err(E::Refused("mission-report-selection-changed")); }
            Ok(())
        })();
        result.map_err(|e| WorkEngineError::Mission(Box::new(crate::MissionSupervisionError::Root(Box::new(e)))))
    }
}
impl RootStore {
    /// Submit one real current worker report under the owning mission lease.
    /// The caller supplies only the retained Agent ID; the engine discovers the
    /// report and rechecks lease/consent/original producer/receipt/settings/bytes
    /// inside the canonical task.submit writer, including before commit. Old
    /// watch generations may finish; manual/active/nonblocked/dialogue work is
    /// not relayed. An existing canonical submission by this producer is not
    /// repeated after quality blocks the task, even across watch generations.
    /// Missing/stale/ambiguous first reports wait without mutation.
    /// Typed errors retain causes and known recorded Work after late cleanup;
    /// no retry, provider, validation, acceptance or author authentication occurs.
    pub fn relay_mission_report(&mut self, lease: &MissionLease, agent_id: &str) -> Result<MissionReportRelayOutcome, E> {
        let work_id = super::mission_supervision_store::bound_mission_work(self, lease)
            .map_err(|e| caused("mission-report-binding", e))?.to_owned();
        let row = self.read_agent(agent_id)?;
        if row.work_id != work_id { return Err(E::Refused("mission-report-work-mismatch")); }
        if row.status != "completed" || row.role != "worker" || (row.native_mission.is_none() && row.origin != "conducteur") { return Ok(MissionReportRelayOutcome::NotApplicable); }
        let work = self.read_work(&work_id)?;
        let task = work.tasks.iter().flatten().find(|t| t.id == row.task_id).ok_or(E::Refused("mission-report-task-missing"))?;
        if task.status != "blocked" || task.brainstorm { return Ok(MissionReportRelayOutcome::NotApplicable); }
        let settings = self.settings()?;
        let chosen = super::material_path::with_material_read(self, || {
            // Original FIFO and durable terminal receipt readers require an
            // actual coherent SQL snapshot even during non-mutating discovery.
            super::mission_supervision_store::read_transaction(&self.db, || {
                let observation = (|| {
                    let current = crate::read_recorded_work(&self.db, &work_id).map_err(E::Work)?;
                    let row = producer(&self.db, &self.directory, &current, agent_id, &settings)?;
                    // A quality-blocked task still owns its canonical handoff.
                    // New watch generations cannot resubmit the same producer,
                    // even if its old report bytes later change.
                    if super::task_validation::latest_submission(&self.db,&current,&row.task_id).map_err(|e|E::Launch(Box::new(e)))?.is_some_and(|r|r.agent_id==agent_id){return Ok(None);}
                    report(&self.root, &row, settings.values.report_clock_skew_seconds().map_err(E::Settings)?).map(Some)
                })();
                observation.map_err(|e| crate::MissionSupervisionError::Root(Box::new(e)))
            }).map_err(|e| caused("mission-report-observation", e))
        })?;
        let chosen = match chosen { None=>return Ok(MissionReportRelayOutcome::NotApplicable),Some(Ok(v))=>v,Some(Err(reason))=>return Ok(MissionReportRelayOutcome::Waiting { reason }) };
        let event = format!("relay-{}", crate::provider_configuration::configuration_digest(format!("{agent_id}/{}/{}",row.attempt_id,chosen.sha256).as_bytes()));
        let request = crate::TaskReportRequest { schema_version: 1, event_id: event, expected_revision: work.revision, id: task.id.clone(), agent_id: agent_id.into(), report_path: chosen.path.clone(), report_sha256: chosen.sha256.clone(), status: "submitted".into() };
        let raw = serde_json::to_vec(&request).map_err(|e| caused("mission-report-request", e))?;
        let root = self.root.clone();
        let fence = RelayFence { lease, directory: self.directory.duplicate().map_err(E::Storage)?, root: &root, agent: agent_id, settings_revision: settings.revision, event_id: &request.event_id, expected_revision: request.expected_revision, report: &chosen };
        let result = self.execute_with_fence(&work_id, "task.submit", &raw, &mut crate::SystemEnvironment, Some(&fence));
        let recorded = result.as_ref().ok().map(|v| (v.result.work.id.clone(),v.result.work.revision));
        let result = finished(result, fence.directory.close().map_err(E::Storage));
        let outcome = match (result, recorded) {
            (Err(primary), Some((work, revision))) => return Err(E::WorkRecorded { primary: Box::new(primary), work, revision }),
            (Err(error), _) => return Err(error),
            (Ok(outcome), _) => outcome,
        };
        Ok(MissionReportRelayOutcome::Submitted { work: outcome.result.work.id, revision: outcome.result.work.revision, agent_id: agent_id.into(), report: chosen.path })
    }
}

// the web UI. It deliberately keeps process exit, report presence and delivery

	State           string `json:"state"`
	Label           string `json:"label"`

	ReportState     string `json:"report_state"`
	ValidationState string `json:"validation_state"`

	ReportID        string `json:"report_id,omitempty"`

}

func submittedReport(t *Task) string {
	if t.Revalidation != nil && strings.TrimSpace(t.Revalidation.Report) != "" {
		return strings.TrimSpace(t.Revalidation.Report)
	}
	const marker = "handoff : "
	if at := strings.LastIndex(t.Next, marker); at >= 0 {
		return strings.TrimSpace(t.Next[at+len(marker):])
	}
	return ""
}

func latestTaskAgent(agents []Agent, task string) *Agent {
	for i := range agents { // Store.agents returns newest first.
		if agents[i].TaskID == task {
			return &agents[i]
		}
	}
	return nil
}

func (s *Store) resultPresentation(w *Work, t *Task, agents []Agent, validation TaskValidation) (p ResultPresentation) {

	// A recovered result can be reviewed without rewriting its producer exit.
	// The review projection checks the current attempt and producer identities.
	if a != nil && (a.Status == "failed" || a.Status == "interrupted") && s.currentReviewPresentation(w, t, a, &p) {
		return p
	}

	if a != nil && (a.Status == "failed" || a.Status == "interrupted") {
		p.State, p.Label = "stopped_early", "Arrêté avant la fin"
		p.ValidationState = "not_validated"
		p.Reason = strings.TrimSpace(a.Activity)
		if p.Reason == "" {
			if a.Status == "failed" {
				p.Reason = "Le processus s’est terminé en échec."
			} else {
				p.Reason = "Le processus a été interrompu."
			}
		}
		if reports := s.taskReports(t.ID); len(reports) > 0 && p.ReportState == "absent" {
			p.ReportState, p.ReportID = "partial_possible", reports[0]
			p.Reason += " Un rapport existe, mais l’arrêt ne permet pas d’en affirmer la complétude."
		}
		p.NextStep = "Examiner le motif, les traces et le rapport éventuel avant de décider d’une reprise."
		return p
	}
	if a != nil && a.Status == "completed" && t.Status == "blocked" && strings.HasPrefix(t.Blocker, "Livraison incomplète :") {
		p.State, p.Label, p.ValidationState = "delivery_incomplete", "Résultat à compléter", "not_validated"
		p.Reason = t.Blocker
		p.NextStep = "Le responsable doit examiner les critères manquants et préparer une correction dans les limites autorisées."
		if report := s.incompleteDeliveryReport(w, t, a); report != "" {
			p.ReportID, p.ReportState = report, "partial_possible"
		}
		return p
	}
	if a != nil && a.Status == "completed" && s.currentReviewPresentation(w, t, a, &p) {
		return p
	}

	if t.Status == "submitted" {
		if a == nil {
			p.ProcessState = "unknown"
		}
		p.ValidationState = "missing_gate"
		p.State, p.Label = "result_to_review", "Résultat à vérifier"
		p.Reason = "Un rapport est soumis ; sa complétude et les critères restent à examiner. Aucune revue de validation n’est enregistrée."
		p.NextStep = "Lire le rapport, vérifier les critères et enregistrer la revue de validation."
		if t.Gate != nil {
			if !s.validGate(t) || len(validation.Blockers) > 0 {
				p.State, p.Label = "validation_withheld", "Validation retenue"
				p.ValidationState = "failed_or_stale"
				p.Reason = "La revue de validation n’autorise pas une validation actuelle."
				if len(validation.Blockers) > 0 {
					p.Reason = validation.Blockers[0]
				}
				p.NextStep = "Corriger ou renouveler les preuves, puis enregistrer une revue de validation actuelle."
			} else {
				p.ValidationState = "ready_for_decision"
				p.Reason = "La revue de validation et ses preuves sont actuelles ; la décision d’acceptation reste à enregistrer."
				p.NextStep = "Examiner le rapport puis enregistrer la décision de validation."
			}
		}
		return p
	}

	if a != nil && a.Status == "completed" {
		p.ProcessState = "completed"
		p.ValidationState = "not_started"
		report, reportReason := s.provenReport(t.ID, a.Started)
		if report != "" {
			p.State, p.Label = "result_to_review", "Résultat à vérifier"
			p.ReportState, p.ReportID = "attributable_unsubmitted", report
			p.Reason = "Le processus est terminé et un rapport lui est attribuable ; sa complétude n’est pas démontrée."
			p.NextStep = "Soumettre le rapport, vérifier les critères et enregistrer la revue de validation."
			return p
		}
		p.State, p.Label = "completed_unproven", "Terminé, résultat non démontré"
		p.ReportState = "absent_or_unattributed"
		p.Reason = "Le processus est terminé normalement, mais " + reportReason + "."
		if strings.TrimSpace(a.Relay) != "" {
			p.Reason = a.Relay
		}
		p.NextStep = "Désigner ou compléter un rapport avant toute validation ou relance."
		return p
	}

	if t.Gate != nil && !s.validGate(t) {
		p.ValidationState = "failed_or_stale"
	}
	return p
}

// provenReport retient le rapport produit par cette tentative : lisible, non
// vide et modifié après son départ. Aucun contenu n'est interprété — la
// lecture du fond reste humaine. Plusieurs candidats restent ambigus : choisir
// à la place de l'opérateur reviendrait à soumettre une preuve non désignée.
func (s *Store) provenReport(task, started string) (string, string) {
	since, e := time.Parse(time.RFC3339Nano, started)
	if e != nil {
		return "", "date de départ de la tentative illisible"
	}
	// Tolérance de dérive : l'horloge murale et la date de modification d'un
	// fichier ne viennent pas de la même source et peuvent se croiser de
	// quelques instants. Une tentative réelle dure bien plus que cette marge,
	// donc elle n'affaiblit pas le refus d'un rapport d'une tentative passée.
	since = since.Add(-reportClockSkew)
	fresh := []string{}
	stale := 0
	for _, rel := range s.taskReports(task) {
		p, e := safeReport(s.root, rel)
		if e != nil {
			continue
		}
		info, e := os.Stat(p)
		if e != nil || info.Size() == 0 {
			continue
		}
		if info.ModTime().Before(since) {
			stale++
			continue
		}
		fresh = append(fresh, rel)
	}
	switch {
	case len(fresh) == 1:
		return fresh[0], ""
	case len(fresh) > 1:
		return "", "plusieurs rapports candidats pour cette tentative (" + strings.Join(fresh, ", ") + ")"
	case stale > 0:
		return "", fmt.Sprintf("aucun rapport produit par cette tentative ; %d rapport(s) antérieur(s) ignoré(s)", stale)
	}
	return "", "aucun rapport lisible et non vide sous docs/ pour cette tâche"
}

func (s *Store) incompleteDeliveryReport(w *Work, t *Task, a *Agent) string {
	if a == nil || a.Status != "completed" || !currentTaskAttempt(t, a.Attempt) || t.Status != "blocked" || !strings.HasPrefix(t.Blocker, "Livraison incomplète :") || w.Planning == nil || w.Planning.Repository == nil {

	if e != nil {
		return ""
	}
	if _, e = safeReport(s.root, rel); e != nil {
		return ""
	}
	return filepath.ToSlash(rel)
}

func safeReport(root, path string) (string, error) {
	p, e := localFile(root, path)
	if e != nil {
		return "", e
	}
	info, e := os.Stat(p)
	if e != nil {
		return "", e
	}
	if !info.Mode().IsRegular() {
		return "", fmt.Errorf("Le rapport doit être un fichier ordinaire")
	}
	return p, nil
}
