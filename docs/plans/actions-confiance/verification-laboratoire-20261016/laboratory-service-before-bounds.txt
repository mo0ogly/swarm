func laboratoryProcessIdentity(pid int) (string, error) {
	raw, err := os.ReadFile(fmt.Sprintf("/proc/%d/stat", pid))
	if err != nil {
		if os.IsNotExist(err) || errors.Is(err, syscall.ESRCH) {
			return "", nil
		}
		return "", err
	}
	parts := strings.SplitN(string(raw), ") ", 2)
	if len(parts) != 2 {
		return "", laboratoryInvalid()
	}
	fields := strings.Fields(parts[1])
	if len(fields) < 20 {
		return "", laboratoryInvalid()
	}
	if fields[0] == "Z" {
		return "", nil
	}
	return fields[19], nil
}
func laboratoryProcessStart(pid int) string { v, _ := laboratoryProcessIdentity(pid); return v }
func laboratoryState(st *trustState) *trustLaboratoryState {
	if st.LaboratoryRuns == nil {
		st.LaboratoryRuns = &trustLaboratoryState{}
	}
	r := st.LaboratoryRuns
	if r.Runs == nil {
		r.Runs = map[string]TrustLaboratoryRun{}
	}
	if r.Events == nil {
		r.Events = map[string]string{}
	}
	if r.Digests == nil {
		r.Digests = map[string]string{}
	}
	return r
}
func validateTrustLaboratoryState(st *trustState) error {
	if err := validateTrustLaboratoryPolicy(st.Laboratory); err != nil {
		return err
	}
	if st.LaboratoryRuns == nil {
		return nil
	}
	r := st.LaboratoryRuns
	for id, v := range r.Runs {
		if id != v.ID || !safeName(id) || !laboratoryStoredInput(&v.Input) || validateTrustLaboratoryPolicy(&v.Policy) != nil || v.Revision < 1 || v.Principal == "" || v.Authority == "" || v.StartedAt.IsZero() || v.OwnerPID < 1 || v.OwnerStart == "" {
			return laboratoryInvalid()
		}
		switch v.Status {
		case "running", "passed", "failed", "invalidated", "interrupted":
		default:
			return laboratoryInvalid()
		}
		if len(v.Corpus) == 0 || hashMust(v.Corpus) != v.Input.Digest {
			return laboratoryInvalid()
		}
		if v.Result != nil && (hashMust(v.Result.Input) != hashMust(v.Input) || v.Result.DeterministicDigest != laboratoryResultDigest(*v.Result) || !v.Result.Simulation) {
			return laboratoryInvalid()
		}
		if v.Status == "passed" && (v.Result == nil || v.FinishedAt == nil) {
			return laboratoryInvalid()
		}
	}
	for event, id := range r.Events {
		if !safeName(event) || r.Runs[id].ID != id || r.Digests[event] == "" {
			return laboratoryInvalid()
		}
	}
	return nil
}
func laboratoryRunView(v TrustLaboratoryRun) TrustLaboratoryRun {
	current, err := laboratoryProcessIdentity(v.OwnerPID)
	if v.Status == "running" && err == nil && current != v.OwnerStart {
		v.Status = "interrupted"
		v.Diagnostic = "owner_lost"
	}
	return v
}
func laboratoryView(st *trustState) any {
	r := laboratoryState(st)
	runs := map[string]TrustLaboratoryRun{}
	active := 0
	for id, run := range r.Runs {
		v := laboratoryRunView(run)
		runs[id] = v
		if v.Status == "running" {
			active++
		}
	}
	return map[string]any{"schema_version": 1, "policy": laboratoryPolicy(st.Laboratory), "configuration_revision": st.Revision, "corpus": laboratoryCorpus(), "corpus_digest": hashMust(laboratoryCorpus()), "runs": runs, "consumption": map[string]int{"active_runs": active, "attempts": len(r.Runs)}, "policy_bounds": trustPolicyBounds(), "production_effects": false, "provider_containment": false, "scope": "fixed_laboratory_corpus_only"}
}
func laboratoryRecord(st *trustState, op string, id trustIdentity, object string, raw []byte, value any, reason string, at time.Time) {
	original := hash(raw)
	st.Originals[original] = append([]byte{}, raw...)
	appendTrustEvidence(st, "laboratory-"+op, id, object, hashMust(value), original, "recorded", reason, at)
}
func (s *Store) laboratoryOperation(ctx context.Context, op string, raw []byte, token string) (any, error) {
	return s.laboratoryOperationUsing(ctx, op, raw, token, func(ctx context.Context, input TrustLaboratoryInput) (TrustLaboratoryResult, error) {
		return runLaboratoryProcess(ctx, input, nil)
	})
}
func (s *Store) laboratoryOperationUsing(ctx context.Context, op string, raw []byte, token string, runner func(context.Context, TrustLaboratoryInput) (TrustLaboratoryResult, error)) (any, error) {
	c, err := parseLaboratory(op, raw)
	if err != nil {
		return nil, err
	}
	var selected TrustLaboratoryRun
	var expectedDeterministic string
	var admitted trustIdentity
	launch := false
	value, err := s.withTrustLock(op != "show" && op != "result" && op != "policy-preview", func(st *trustState) (any, error) {
		now := time.Now().UTC()
		id, e := findTrustIdentityAt(st, token, now)
		if e != nil || !trustSecretAdmin(id) || ctx.Err() != nil {
			return nil, laboratoryDenied()
		}
		if op == "show" {
			return laboratoryView(st), nil
		}
		r := laboratoryState(st)
		if op == "result" {
			v, ok := r.Runs[c.RunID]
			if !ok {
				return nil, laboratoryInvalid()
			}
			return laboratoryRunView(v), nil
		}
		if op == "policy-apply" {
			event := "laboratory:" + c.EventID
			if prior, ok := st.EventDigests[event]; ok {
				if prior != hashMust(c) || st.EventAuthorities[event] != trustAuthorityDigest(id) {
					return nil, laboratoryDenied()
				}
				var v any
				if json.Unmarshal(st.EventReplies[event], &v) != nil {
					return nil, laboratoryInvalid()
				}
				return v, nil
			}
		}
		if c.ExpectedRevision == nil || *c.ExpectedRevision != st.Revision {
			return nil, &CommandError{Code: "revision_conflict", Message: "configuration modifiée ; rechargez avant de confirmer"}
		}
		if op == "policy-preview" || op == "policy-apply" {
			if c.Policy == nil || validateTrustLaboratoryPolicy(c.Policy) != nil {
				return nil, laboratoryInvalid()
			}
			if op == "policy-preview" {
				return map[string]any{"current_revision": st.Revision, "proposed_revision": st.Revision + 1, "policy": c.Policy, "applied": false}, nil
			}
			event := "laboratory:" + c.EventID
			digest := hashMust(c)
			if prior, ok := st.EventDigests[event]; ok {
				if prior != digest {
					return nil, laboratoryInvalid()
				}
				var v any
				if json.Unmarshal(st.EventReplies[event], &v) != nil {
					return nil, laboratoryInvalid()
				}
				return v, nil
			}
			st.Revision++
			st.Laboratory = c.Policy
			st.OIDCPending = nil
			st.OIDCSessions = nil
			st.WorkloadSessions = nil
			st.History = append(st.History, trustConfigRevision{Revision: st.Revision, Policy: st.Policy, Actor: id.Subject, Reason: c.Reason, EventID: c.EventID, At: now.Format(time.RFC3339Nano)})
			v := map[string]any{"revision": st.Revision, "policy": c.Policy, "applied": true}
			b, _ := json.Marshal(v)
			if st.EventAuthorities == nil {
				st.EventAuthorities = map[string]string{}
			}
			st.EventAuthorities[event] = trustAuthorityDigest(id)
			st.EventDigests[event] = digest
			st.EventReplies[event] = b
			laboratoryRecord(st, op, id, "laboratory-policy", raw, v, c.Reason, now)
			return v, nil
		}
		p := laboratoryPolicy(st.Laboratory)
		if !bool(p.Enabled) {
			return nil, laboratoryDenied()
		}
		digest := hashMust(struct {
			Operation string
			Command   TrustLaboratoryCommand
		}{op, c})
		if prior, ok := r.Events[c.EventID]; ok {
			if r.Digests[c.EventID] != digest {
				return nil, &CommandError{Code: "event_conflict", Message: "événement de laboratoire réutilisé avec un autre contenu"}
			}
			v := r.Runs[prior]
			if v.Principal != id.Subject || v.Authority != trustAuthorityDigest(id) {
				return nil, laboratoryDenied()
			}
			return laboratoryRunView(v), nil
		}
		input := c.Input
		parent := ""
		if op == "replay" {
			old, ok := r.Runs[c.RunID]
			old = laboratoryRunView(old)
			if !ok || old.Status == "running" {
				return nil, laboratoryInvalid()
			}
			if old.Result != nil {
				expectedDeterministic = old.Result.DeterministicDigest
			}
			copy := old.Input
			input = &copy
			parent = old.ID
		}
		if !laboratoryInputValid(input) {
			return nil, laboratoryInvalid()
		}
		active := 0
		for key, run := range r.Runs {
			view := laboratoryRunView(run)
			if view.Status == "running" {
				active++
			} else if run.Status == "running" {
				run.Status = "interrupted"
				run.Diagnostic = "owner_lost"
				run.FinishedAt = &now
				r.Runs[key] = run
			}
		}
		if active >= p.MaxConcurrentRuns {
			return nil, &CommandError{Code: "operation_pending", Message: "laboratoire occupé : concurrence autorisée atteinte"}
		}
		owner := laboratoryProcessStart(os.Getpid())
		if owner == "" {
			return nil, laboratoryInvalid()
		}
		corpus, _ := json.Marshal(laboratoryCorpus())
		selected = TrustLaboratoryRun{Corpus: corpus, ID: "lab-" + digest[:32], Parent: parent, Input: *input, Policy: p, Revision: st.Revision, Principal: id.Subject, Authority: trustAuthorityDigest(id), Status: "running", StartedAt: now, OwnerPID: os.Getpid(), OwnerStart: owner}
		if _, exists := r.Runs[selected.ID]; exists {
			return nil, laboratoryInvalid()
		}
		r.Runs[selected.ID] = selected
		r.Events[c.EventID] = selected.ID
		r.Digests[c.EventID] = digest
		laboratoryRecord(st, "admit", id, selected.ID, raw, selected, c.Reason, now)
		admitted = id
		launch = true
		return selected, nil
	})
	if err != nil || !launch {
		return value, err
	}
	limited, cancel := context.WithTimeout(ctx, time.Duration(selected.Policy.TimeoutSeconds)*time.Second)
	defer cancel()
	result, executionErr := runner(limited, selected.Input)
	if limited.Err() != nil {
		executionErr = limited.Err()
	}
	if executionErr == nil {
		wire, _ := json.Marshal(result)
		result, executionErr = laboratoryPublicResult(wire, selected.Input)
	}
	if executionErr == nil && expectedDeterministic != "" && result.DeterministicDigest != expectedDeterministic {
		executionErr = laboratoryInvalid()
	}
	var rejected error
	final, err := s.withTrustLock(true, func(st *trustState) (any, error) {
		now := time.Now().UTC()
		r := laboratoryState(st)
		run, ok := r.Runs[selected.ID]
		if !ok || run.Status != "running" || run.Authority != selected.Authority {
			return nil, laboratoryInvalid()
		}
		id, e := findTrustIdentityAt(st, token, now)
		if e != nil || !trustSecretAdmin(id) || trustAuthorityDigest(id) != selected.Authority || st.Revision != selected.Revision || laboratoryPolicy(st.Laboratory) != selected.Policy || ctx.Err() != nil {
			run.Status = "invalidated"
			run.Diagnostic = "authority_or_revision_changed"
			rejected = laboratoryDenied()
		}
		if run.Status == "running" {
			if executionErr != nil {
				run.Status = "failed"
				run.Diagnostic = "isolated_execution_failed"
				rejected = &CommandError{Code: "laboratory_failed", Message: "laboratoire : exécution isolée échouée ou interrompue ; consulter la tentative"}
			} else {
				run.Status = "passed"
				run.Result = &result
			}
		}
		run.FinishedAt = &now
		r.Runs[run.ID] = run
		actor := id
		if e != nil || actor.Subject != selected.Principal {
			actor = admitted
		}
		finalRaw, _ := json.Marshal(map[string]any{"run_id": run.ID, "status": run.Status, "diagnostic_code": run.Diagnostic})
		laboratoryRecord(st, "finish", actor, run.ID, finalRaw, run, "fin de tentative isolée sans effet de production", now)
		return run, nil
	})
	if err != nil {
		return nil, err
	}
	if rejected != nil {
		return nil, rejected
	}
	return final, nil
}
