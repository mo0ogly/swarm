//go:build linux

package engine

import (
	"context"
	"crypto/ed25519"
	"crypto/rand"
	"encoding/base64"
	"encoding/json"
	"time"
)

func (r *trustEvidenceRuntime) purge(revision int) {
	if r.revision != revision {
		for id, k := range r.keys {
			clear(k)
			delete(r.keys, id)
		}
		r.revision = revision
	}
	if r.keys == nil {
		r.keys = map[string]ed25519.PrivateKey{}
	}
}
func evidenceRegistry(st *trustState) *trustEvidenceRegistry {
	if st.PublicEvidence == nil {
		st.PublicEvidence = &trustEvidenceRegistry{}
	}
	r := st.PublicEvidence
	if r.Keys == nil {
		r.Keys = map[string]TrustEvidenceKey{}
	}
	if r.Replies == nil {
		r.Replies = map[string]trustEvidenceReply{}
	}
	return r
}
func evidenceExport(st *trustState) TrustProofExport {
	originals := map[string][]byte{}
	for _, ev := range st.Evidence {
		originals[ev.OriginalHash] = append([]byte{}, st.Originals[ev.OriginalHash]...)
	}
	return TrustProofExport{Schema: 2, Simulation: true, Evidence: append([]TrustEvidence{}, st.Evidence...), Originals: originals}
}
func (s *Store) trustEvidenceOperation(ctx context.Context, op string, raw []byte, token string, persistent bool) (any, error) {
	cmd, err := parseTrustEvidence(op, raw)
	if err != nil {
		return nil, err
	}
	if op == "verify" {
		return verifyPublicEvidence(cmd)
	}
	var prepared ed25519.PrivateKey
	if op == "key-prepare" {
		if !persistent || s.evidenceRuntime == nil {
			return nil, trustEvidenceDenied()
		}
		_, err = s.withTrustReadAvailable(func(st *trustState) (any, error) {
			id, e := findTrustIdentityAt(st, token, time.Now().UTC())
			if e != nil || !trustSecretAdmin(id) || !bool(trustEvidencePolicy(st.EvidenceSigning).Enabled) {
				return nil, trustEvidenceDenied()
			}
			return nil, nil
		})
		if err != nil {
			return nil, err
		}
		if err = trustSecretHarden(); err != nil {
			return nil, err
		}
		_, prepared, err = ed25519.GenerateKey(rand.Reader)
		if err != nil {
			return nil, err
		}
		defer func() { clear(prepared) }()
	}
	var storeKey string
	var expires time.Time
	var removeKeys []string
	write := op != "show" && op != "policy-preview" && op != "anchor"
	result, err := s.withTrustLock(write, func(st *trustState) (any, error) {
		now := time.Now().UTC()
		id, e := findTrustIdentityAt(st, token, now)
		if ctx.Err() != nil || e != nil || !trustSecretAdmin(id) {
			return nil, trustEvidenceDenied()
		}
		p := trustEvidencePolicy(st.EvidenceSigning)
		r := evidenceRegistry(st)
		b := s.evidenceRuntime
		if b == nil {
			return nil, trustEvidenceDenied()
		}
		b.mu.Lock()
		defer b.mu.Unlock()
		b.purge(st.Revision)
		if op == "show" {
			available := map[string]bool{}
			for key, k := range r.Keys {
				available[key] = len(b.keys[key]) == ed25519.PrivateKeySize && now.Before(k.NotAfter) && k.PolicyRevision == st.Revision
			}
			seq, last := 0, ""
			if len(st.Evidence) > 0 {
				ev := st.Evidence[len(st.Evidence)-1]
				seq, last = ev.Sequence, ev.Hash
			}
			return map[string]any{"schema_version": 1, "policy": p, "configuration_revision": st.Revision, "registry_revision": r.Revision, "active_key_id": r.Active, "pending_key_id": r.Pending, "keys": r.Keys, "material_available": available, "last_sequence": seq, "last_hash": last, "policy_bounds": trustPolicyBounds(), "private_keys_persisted": false}, nil
		}
		if op == "anchor" {
			if !persistent || !bool(p.Enabled) || cmd.ExpectedRegistryRevision == nil || *cmd.ExpectedRegistryRevision != r.Revision || cmd.KeyID != r.Active || r.Active == "" {
				return nil, trustEvidenceDenied()
			}
			k := r.Keys[r.Active]
			priv := b.keys[r.Active]
			if k.State != "active" || k.PolicyRevision != st.Revision || !now.Before(k.NotAfter) || len(priv) != ed25519.PrivateKeySize || hash(priv.Public().(ed25519.PublicKey)) != k.ID || len(st.Evidence) == 0 {
				return nil, trustEvidenceDenied()
			}
			last := st.Evidence[len(st.Evidence)-1]
			if cmd.ExpectedSequence == nil || *cmd.ExpectedSequence != last.Sequence || cmd.ExpectedHash != last.Hash {
				return nil, &CommandError{Code: "revision_conflict", Message: "preuve modifiée ; rechargez la séquence et l’empreinte avant signature"}
			}
			export := evidenceExport(st)
			if e := verifyTrustOriginals(export); e != nil {
				return nil, e
			}
			a := TrustPublicAnchor{Schema: 3, Domain: "swarm-public-evidence-anchor-v1", KeyID: k.ID, RegistryRevision: r.Revision, PolicyRevision: st.Revision, Sequence: last.Sequence, Hash: last.Hash, ExportDigest: hashMust(export), SignedAt: now}
			a.Signature = base64.RawURLEncoding.EncodeToString(ed25519.Sign(priv, publicAnchorBytes(a)))
			return map[string]any{"export": export, "anchor": a}, nil
		}
		if op != "policy-preview" && op != "policy-apply" && !bool(p.Enabled) {
			return nil, trustEvidenceDenied()
		}
		digest := hashMust(struct {
			Operation string
			Command   TrustEvidenceCommand
		}{op, cmd})
		if prior, ok := r.Replies[cmd.EventID]; ok {
			if prior.Digest != digest {
				return nil, &CommandError{Code: "event_conflict", Message: "événement de preuve réutilisé avec un autre contenu"}
			}
			if prior.Authority != trustAuthorityDigest(id) {
				return nil, trustEvidenceDenied()
			}
			var value any
			if json.Unmarshal(prior.Value, &value) != nil {
				return nil, trustEvidenceInvalid()
			}
			return value, nil
		}
		var value any
		var object string
		if op == "policy-preview" || op == "policy-apply" {
			if cmd.Policy == nil || cmd.ExpectedRevision == nil || *cmd.ExpectedRevision != st.Revision {
				return nil, &CommandError{Code: "revision_conflict", Message: "configuration modifiée ; rechargez avant de confirmer"}
			}
			if err := validateTrustEvidencePolicy(cmd.Policy); err != nil {
				return nil, err
			}
			if op == "policy-preview" {
				return map[string]any{"current_revision": st.Revision, "proposed_revision": st.Revision + 1, "policy": cmd.Policy, "applied": false}, nil
			}
			st.Revision++
			st.EvidenceSigning = cmd.Policy
			st.OIDCPending = nil
			st.OIDCSessions = nil
			st.WorkloadSessions = nil
			st.History = append(st.History, trustConfigRevision{Revision: st.Revision, Policy: st.Policy, Actor: id.Subject, Reason: cmd.Reason, EventID: cmd.EventID, At: now.Format(time.RFC3339Nano)})
			value = map[string]any{"revision": st.Revision, "policy": cmd.Policy, "applied": true}
			object = "evidence-policy"
		} else {
			if !persistent || cmd.ExpectedRegistryRevision == nil || *cmd.ExpectedRegistryRevision != r.Revision || int64(r.Revision) >= trustPolicyBounds()["evidence_registry_revision"]["max"] {
				return nil, &CommandError{Code: "revision_conflict", Message: "registre de clés modifié ; rechargez avant confirmation"}
			}
			var k TrustEvidenceKey
			switch op {
			case "key-prepare":
				if r.Pending != "" {
					old := r.Keys[r.Pending]
					if len(b.keys[old.ID]) == ed25519.PrivateKeySize && now.Before(old.NotAfter) && old.PolicyRevision == st.Revision {
						return nil, &CommandError{Code: "operation_pending", Message: "une clé attend déjà son second administrateur"}
					}
					old.State = "abandoned"
					r.Keys[old.ID] = old
					removeKeys = append(removeKeys, old.ID)
				}
				if len(prepared) != ed25519.PrivateKeySize {
					return nil, trustEvidenceDenied()
				}
				pub := prepared.Public().(ed25519.PublicKey)
				k = TrustEvidenceKey{ID: hash(pub), PublicKey: base64.RawURLEncoding.EncodeToString(pub), Proposer: id.Subject, PolicyRevision: st.Revision, CreatedAt: now, NotAfter: now.Add(time.Duration(p.KeyTTLSeconds) * time.Second), State: "pending"}
				k.Digest = evidenceKeyDigest(k)
				r.Pending = k.ID
				storeKey = k.ID
				expires = k.NotAfter
			case "key-activate", "key-revoke":
				var ok bool
				k, ok = r.Keys[cmd.KeyID]
				if !ok || k.Digest != cmd.KeyDigest {
					return nil, trustEvidenceDenied()
				}
				if op == "key-activate" {
					priv := b.keys[k.ID]
					if k.State != "pending" || r.Pending != k.ID || k.PolicyRevision != st.Revision || id.Subject == k.Proposer || !now.Before(k.NotAfter) || len(priv) != ed25519.PrivateKeySize || hash(priv.Public().(ed25519.PublicKey)) != k.ID {
						return nil, trustEvidenceDenied()
					}
					if r.Active != "" {
						old := r.Keys[r.Active]
						old.State = "retired"
						old.RetiredAt = &now
						r.Keys[old.ID] = old
						removeKeys = append(removeKeys, old.ID)
					}
					k.State = "active"
					k.ActivatedBy = id.Subject
					k.ActivatedAt = &now
					r.Active = k.ID
					r.Pending = ""
				} else {
					if k.State == "revoked" || k.State == "abandoned" {
						return nil, trustEvidenceDenied()
					}
					k.State = "revoked"
					k.RevokedAt = &now
					if r.Active == k.ID {
						r.Active = ""
					}
					if r.Pending == k.ID {
						r.Pending = ""
					}
					removeKeys = append(removeKeys, k.ID)
				}
			default:
				return nil, trustEvidenceInvalid()
			}
			r.Revision++
			r.Keys[k.ID] = k
			value = map[string]any{"key": k, "registry_revision": r.Revision}
			object = k.ID
		}
		if err := verifyTrustOriginals(evidenceExport(st)); err != nil {
			return nil, err
		}
		original := hash(raw)
		st.Originals[original] = append([]byte{}, raw...)
		appendTrustEvidence(st, "evidence-"+op, id, object, hashMust(value), original, "applied", cmd.Reason, now)
		reply, _ := json.Marshal(value)
		r.Replies[cmd.EventID] = trustEvidenceReply{Digest: digest, Authority: trustAuthorityDigest(id), Value: reply}
		return value, nil
	})
	if err != nil {
		return nil, err
	}
	if storeKey != "" || len(removeKeys) > 0 {
		b := s.evidenceRuntime
		b.mu.Lock()
		for _, id := range removeKeys {
			clear(b.keys[id])
			delete(b.keys, id)
		}
		if storeKey != "" {
			key := result.(map[string]any)["key"].(TrustEvidenceKey)
			if b.revision == key.PolicyRevision {
				b.keys[storeKey] = append(ed25519.PrivateKey{}, prepared...)
			}
			result.(map[string]any)["material_available"] = len(b.keys[storeKey]) == ed25519.PrivateKeySize
		}
		b.mu.Unlock()
		if storeKey != "" {
			time.AfterFunc(time.Until(expires), func() { b.mu.Lock(); defer b.mu.Unlock(); clear(b.keys[storeKey]); delete(b.keys, storeKey) })
		}
	}
	return result, nil
}
